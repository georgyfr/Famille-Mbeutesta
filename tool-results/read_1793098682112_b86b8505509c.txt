     1→"use client";
     2→
     3→import { useCallback, useEffect, useRef, useState } from "react";
     4→import { toast } from "sonner";
     5→import { startAuthentication, startRegistration } from "@simplewebauthn/browser";
     6→import { api, deviceName, ensureDeviceSuffix } from "@/lib/fc-client";
     7→import { FcButton, FcCard, FcDivider, FcInput, FcLogo, FcSpinner, FcTitle, FcWatermark } from "./ui";
     8→import { Fingerprint, KeyRound, LifeBuoy, Lock, ShieldCheck, ArrowLeft, MailCheck, Copy, Check, RefreshCw } from "lucide-react";
     9→
    10→type MemberInfo = { fullName: string; branch: string; role: string; phone: string | null; reenroll: boolean };
    11→
    12→type Step =
    13→  | { k: "welcome" }
    14→  | { k: "code-entry" }
    15→  | { k: "claim-confirm"; claimToken: string; member: MemberInfo }
    16→  | { k: "otp"; claimToken: string; member: MemberInfo; demoCode?: string }
    17→  | { k: "recovery-entry" }
    18→  | { k: "pin-login"; uid: string; name: string; hasPasskey: boolean }
    19→  | { k: "setup"; recoveryCode: string; userName: string; reenroll: boolean };
    20→
    21→export function AuthFlow({ onReady }: { onReady: (name: string) => void }) {
    22→  const [step, setStep] = useState<Step>({ k: "welcome" });
    23→  const [busy, setBusy] = useState(false);
    24→  const [remembered, setRemembered] = useState<{ uid: string; name: string; hasPasskey: boolean } | null>(null);
    25→
    26→  useEffect(() => {
    27→    const t = setTimeout(() => {
    28→      const uid = localStorage.getItem("fc_uid");
    29→      const name = localStorage.getItem("fc_name");
    30→      if (uid && name) {
    31→        setRemembered({ uid, name, hasPasskey: localStorage.getItem("fc_has_passkey") === "1" });
    32→      }
    33→      ensureDeviceSuffix();
    34→    }, 0);
    35→    return () => clearTimeout(t);
    36→  }, []);
    37→
    38→  return (
    39→    <div className="relative flex min-h-full flex-col items-center px-4 py-8 sm:py-12">
    40→      <FcWatermark />
    41→
    42→      {/* En-tête de marque — fidèle à la maquette */}
    43→      <div className="relative z-10 w-full max-w-md rounded-b-[40px] pb-6 pt-2">
    44→        <FcLogo size={132} />
    45→      </div>
    46→
    47→      <div className="relative z-10 w-full max-w-md space-y-4">
    48→        {step.k === "welcome" && (
    49→          <WelcomeScreen
    50→            remembered={remembered}
    51→            busy={busy}
    52→            setBusy={setBusy}
    53→            onReady={onReady}
    54→            goto={setStep}
    55→          />
    56→        )}
    57→        {step.k === "code-entry" && (
    58→          <CodeEntryScreen busy={busy} setBusy={setBusy} goto={setStep} />
    59→        )}
    60→        {step.k === "claim-confirm" && (
    61→          <ClaimConfirmScreen
    62→            state={step}
    63→            busy={busy}
    64→            setBusy={setBusy}
    65→            goto={setStep}
    66→          />
    67→        )}
    68→        {step.k === "otp" && (
    69→          <OtpScreen state={step} busy={busy} setBusy={setBusy} goto={setStep} onReady={onReady} />
    70→        )}
    71→        {step.k === "recovery-entry" && (
    72→          <RecoveryEntryScreen busy={busy} setBusy={setBusy} goto={setStep} onReady={onReady} />
    73→        )}
    74→        {step.k === "pin-login" && (
    75→          <PinLoginScreen state={step} busy={busy} setBusy={setBusy} goto={setStep} onReady={onReady} />
    76→        )}
    77→        {step.k === "setup" && (
    78→          <SetupScreen state={step} busy={busy} setBusy={setBusy} onReady={onReady} />
    79→        )}
    80→      </div>
    81→    </div>
    82→  );
    83→}

    85→/* ───────────────────────── Écran d'accueil ───────────────────────── */
    86→
    87→function WelcomeScreen({
    88→  remembered,
    89→  busy,
    90→  setBusy,
    91→  onReady,
    92→  goto,
    93→}: {
    94→  remembered: { uid: string; name: string; hasPasskey: boolean } | null;
    95→  busy: boolean;
    96→  setBusy: (b: boolean) => void;
    97→  onReady: (name: string) => void;
    98→  goto: (s: Step) => void;
    99→}) {
   100→  const handleRealPasskey = useCallback(async () => {
   101→    if (!remembered) return;
   102→    setBusy(true);
   103→    try {
   104→      const { options } = await api<{ options: unknown }>("/api/auth/passkey", {
   105→        body: { action: "auth-options" },
   106→      });
   107→      const response = await startAuthentication({ optionsJSON: options as never });
   108→      const result = await api<{ user: { fullName: string } }>("/api/auth/passkey", {
   109→        body: { action: "auth-verify", response },
   110→      });
   111→      if (!result.ok) throw new Error(result.error);
   112→      localStorage.setItem("fc_uid", remembered.uid);
   113→      localStorage.setItem("fc_name", result.user.fullName);
   114→      toast.success(`Content de vous revoir, ${result.user.fullName.split(" ")[0]} !`);
   115→      onReady(result.user.fullName);
   116→    } catch (e) {
   117→      toast.error(e instanceof Error ? e.message : "Connexion biométrique impossible sur cet appareil.");
   118→    } finally {
   119→      setBusy(false);
   120→    }
   121→  }, [remembered, onReady, setBusy]);
   122→
   123→  const handleSimulatedPasskey = useCallback(async () => {
   124→    if (!remembered) return;
   125→    setBusy(true);
   126→    try {
   127→      const result = await api<{ user: { fullName: string } }>("/api/auth/passkey", {
   128→        body: { action: "simulate", uid: remembered.uid },
   129→      });
   130→      if (!result.ok) throw new Error(result.error);
   131→      localStorage.setItem("fc_uid", remembered.uid);
   132→      localStorage.setItem("fc_name", result.user.fullName);
   133→      toast.success(`Content de vous revoir, ${result.user.fullName.split(" ")[0]} ! (biométrie simulée)`);
   134→      onReady(result.user.fullName);
   135→    } catch (e) {
   136→      toast.error(e instanceof Error ? e.message : "Échec de la connexion simulée.");
   137→    } finally {
   138→      setBusy(false);
   139→    }
   140→  }, [remembered, onReady, setBusy]);
   141→
   142→  return (
   143→    <FcCard className="p-6 sm:p-8">
   144→      <FcTitle className="mb-1">FamilleConnect</FcTitle>
   145→      <p className="mb-6 text-center text-[15px] text-[#5C6B60]">
   146→        Le registre vivant de la Famille Mbeutetsa
   147→      </p>
   148→
   149→      {remembered && (
   150→        <div className="mb-5 rounded-2xl border border-[#C89F62]/40 bg-[#C89F62]/10 p-4 text-center">
   151→          <p className="text-sm text-[#7A5B2E]">
   152→            Bon retour parmi nous, <span className="font-bold">{remembered.name}</span>
   153→          </p>
   154→        </div>
   155→      )}
   156→
   157→      {remembered && remembered.hasPasskey && (
   158→        <div className="mb-3 space-y-2">
   159→          <FcButton onClick={handleRealPasskey} disabled={busy}>
   160→            {busy ? <FcSpinner /> : <Fingerprint className="h-5 w-5" />}
   161→            Se connecter avec mon empreinte
   162→          </FcButton>
   163→          <FcButton variant="ghost" className="text-sm" onClick={handleSimulatedPasskey} disabled={busy}>
   164→            <ShieldCheck className="h-4 w-4" /> Biométrie simulée (démo)
   165→          </FcButton>
   166→          <FcDivider label="ou" />
   167→        </div>
   168→      )}
   169→
   170→      {remembered && (
   171→        <div className="mb-3 space-y-2">
   172→          <FcButton
   173→            variant="outline"
   174→            onClick={() => goto({ k: "pin-login", uid: remembered.uid, name: remembered.name, hasPasskey: remembered.hasPasskey })}
   175→          >
   176→            <Lock className="h-5 w-5" /> Mon code PIN
   177→          </FcButton>
   178→          <FcDivider label="nouvel appareil" />
   179→        </div>
   180→      )}
   181→
   182→      <div className="space-y-2">
   183→        <FcButton onClick={() => goto({ k: "code-entry" })} disabled={busy}>
   184→          <KeyRound className="h-5 w-5" /> J&apos;ai un code d&apos;accès
   185→        </FcButton>
   186→        <FcButton variant="ghost" className="text-sm" onClick={() => goto({ k: "recovery-entry" })} disabled={busy}>
   187→          <LifeBuoy className="h-4 w-4" /> Récupérer mon accès (code personnel)
   188→        </FcButton>
   189→      </div>
   190→
   191→      <p className="mt-5 rounded-xl bg-[#3D6B52]/5 p-3 text-center text-[13px] leading-relaxed text-[#5C6B60]">
   192→        Pas encore de code ? Le <strong>secrétaire de famille</strong> pré-enregistre chaque membre et remet
   193→        un code d&apos;accès. Rendez-vous à la prochaine réunion, ou contactez-le.
   194→      </p>
   195→    </FcCard>
   196→  );
   197→}

   199→/* ───────────────────────── Saisie du code d'accès ───────────────────────── */
   200→
   201→function CodeEntryScreen({
   202→  busy,
   203→  setBusy,
   204→  goto,
   205→}: {
   206→  busy: boolean;
   207→  setBusy: (b: boolean) => void;
   208→  goto: (s: Step) => void;
   209→}) {
   210→  const [code, setCode] = useState("");
   211→  const handle = async () => {
   212→    if (code.trim().length < 4) {
   213→      toast.error("Entrez le code d'accès remis par le secrétaire (ex : FC-XXXX-XXXX).");
   214→      return;
   215→    }
   216→    setBusy(true);
   217→    try {
   218→      const res = await api<{ claimToken: string; member: MemberInfo }>("/api/auth/verify-code", {
   219→        body: { code, deviceName: deviceName() },
   220→      });
   221→      if (!res.ok) throw new Error(res.error);
   222→      toast.success("Code validé ✓");
   223→      goto({ k: "claim-confirm", claimToken: res.claimToken, member: res.member });
   224→    } catch (e) {
   225→      toast.error(e instanceof Error ? e.message : "Code invalide.");
   226→    } finally {
   227→      setBusy(false);
   228→    }
   229→  };
   230→  return (
   231→    <FcCard className="p-6 sm:p-8">
   232→      <BackButton onClick={() => goto({ k: "welcome" })} />
   233→      <h2 className="fc-serif mb-1 text-center text-2xl font-bold text-[#1E4D2B]">Code d&apos;accès</h2>
   234→      <p className="mb-6 text-center text-[14px] text-[#5C6B60]">
   235→        Saisissez le code remis par le secrétaire de famille.
   236→      </p>
   237→      <FcInput
   238→        icon={<KeyRound className="h-5 w-5" />}
   239→        placeholder="FC-XXXX-XXXX"
   240→        value={code}
   241→        onChange={(e) => setCode(e.target.value.toUpperCase())}
   242→        onKeyDown={(e) => e.key === "Enter" && handle()}
   243→        autoCapitalize="characters"
   244→        autoComplete="one-time-code"
   245→        className="mb-4"
   246→        aria-label="Code d'accès"
   247→      />
   248→      <FcButton onClick={handle} disabled={busy}>
   249→        {busy ? <FcSpinner /> : "Vérifier mon code"}
   250→      </FcButton>
   251→    </FcCard>
   252→  );
   253→}

   311→/* ───────────────────────── Vérification OTP (mode démo) ───────────────────────── */
   312→
   313→function OtpScreen({
   314→  state,
   315→  busy,
   316→  setBusy,
   317→  goto,
   318→  onReady,
   319→}: {
   320→  state: Extract<Step, { k: "otp" }>;
   321→  busy: boolean;
   322→  setBusy: (b: boolean) => void;
   323→  goto: (s: Step) => void;
   324→  onReady: (name: string) => void;
   325→}) {
   326→  const [code, setCode] = useState("");
   327→  const [demoCode, setDemoCode] = useState<string | undefined>(state.demoCode);
   328→  const requested = useRef(false);
   329→
   330→  const requestCode = useCallback(async () => {
   331→    setBusy(true);
   332→    try {
   333→      const res = await api<{ demoMode: boolean; demoCode?: string }>("/api/auth/otp", {
   334→        body: { action: "request", claimToken: state.claimToken },
   335→      });
   336→      if (!res.ok) throw new Error(res.error);
   337→      if (res.demoCode) {
   338→        setDemoCode(res.demoCode);
   339→        toast.info(`Mode démo — votre code de vérification : ${res.demoCode}`);
   340→      } else {
   341→        toast.success("Code de vérification envoyé par SMS.");
   342→      }
   343→    } catch (e) {
   344→      toast.error(e instanceof Error ? e.message : "Envoi du code impossible.");
   345→    } finally {
   346→      setBusy(false);
   347→    }
   348→  }, [state.claimToken, setBusy]);
   349→
   350→  useEffect(() => {
   351→    if (!requested.current) {
   352→      requested.current = true;
   353→      requestCode();
   354→    }
   355→  }, [requestCode]);
   356→
   357→  const verify = async () => {
   358→    setBusy(true);
   359→    try {
   360→      const res = await api<{ activateToken: string }>("/api/auth/otp", {
   361→        body: { action: "verify", claimToken: state.claimToken, code },
   362→      });
   363→      if (!res.ok) throw new Error(res.error);
   364→      const activation = await api<{ recoveryCode: string; user: { fullName: string }; reenroll: boolean }>(
   365→        "/api/auth/activate",
   366→        { body: { activateToken: res.activateToken } }
   367→      );
   368→      if (!activation.ok) throw new Error(activation.error);
   369→      localStorage.setItem("fc_uid", activation.user.id);
   370→      localStorage.setItem("fc_name", activation.user.fullName);
   371→      localStorage.removeItem("fc_has_passkey");
   372→      toast.success(activation.reenroll ? "Accès réactivé ✓" : "Compte activé ✓ Bienvenue !");
   373→      goto({ k: "setup", recoveryCode: activation.recoveryCode, userName: activation.user.fullName, reenroll: activation.reenroll });
   374→    } catch (e) {
   375→      toast.error(e instanceof Error ? e.message : "Code incorrect.");
   376→    } finally {
   377→      setBusy(false);
   378→    }
   379→  };
   380→
   381→  return (
   382→    <FcCard className="p-6 sm:p-8">
   383→      <BackButton onClick={() => goto({ k: "code-entry" })} />
   384→      <h2 className="fc-serif mb-1 text-center text-2xl font-bold text-[#1E4D2B]">Vérification</h2>
   385→      <p className="mb-5 text-center text-[14px] text-[#5C6B60]">
   386→        Entrez le code à 6 chiffres envoyé pour {state.member.fullName}.
   387→      </p>
   388→
   389→      {demoCode && (
   390→        <div className="mb-4 rounded-2xl border border-dashed border-[#C89F62] bg-[#C89F62]/10 p-4 text-center">
   391→          <p className="text-xs font-semibold uppercase tracking-wide text-[#7A5B2E]">Mode démo — votre code</p>
   392→          <p className="fc-serif text-3xl font-bold tracking-[0.35em] text-[#1E4D2B]">{demoCode}</p>
   393→        </div>
   394→      )}
   395→
   396→      <FcInput
   397→        icon={<MailCheck className="h-5 w-5" />}
   398→        placeholder="● ● ● ● ● ●"
   399→        value={code}
   400→        onChange={(e) => setCode(e.target.value.replace(/\D/g, "").slice(0, 6))}
   401→        onKeyDown={(e) => e.key === "Enter" && code.length === 6 && verify()}
   402→        inputMode="numeric"
   403→        autoComplete="one-time-code"
   404→        maxLength={6}
   405→        className="mb-4 text-center tracking-[0.4em]"
   406→        aria-label="Code de vérification à 6 chiffres"
   407→      />
   408→      <FcButton onClick={verify} disabled={busy || code.length !== 6}>
   409→        {busy ? <FcSpinner /> : "Vérifier"}
   410→      </FcButton>
   411→      <button
   412→        className="mt-3 flex w-full items-center justify-center gap-1.5 text-sm text-[#C89F62] hover:underline"
   413→        onClick={() => {
   414→          setCode("");
   415→          requestCode();
   416→        }}
   417→        disabled={busy}
   418→      >
   419→        <RefreshCw className="h-3.5 w-3.5" /> Renvoyer un nouveau code
   420→      </button>
   421→    </FcCard>
   422→  );
   423→}

   495→  state: Extract<Step, { k: "pin-login" }>;

   497→  setBusy: (b: boolean) => void;

   504→    try {
   505→      const res = await api<{ user: { id: string; fullName: string } }>("/api/auth/pin", {
   506→        body: { action: "verify", uid: state.uid, pin, deviceName: deviceName() },
   507→      });
   508→      if (!res.ok) throw new Error(res.error);
   509→      localStorage.setItem("fc_uid", res.user.id);
   510→      localStorage.setItem("fc_name", res.user.fullName);
   511→      toast.success(`Content de vous revoir, ${res.user.fullName.split(" ")[0]} !`);
   512→      onReady(res.user.fullName);
   513→    } catch (e) {
   514→      toast.error(e instanceof Error ? e.message : "PIN incorrect.");
   515→      setPin("");
   516→    } finally {
   517→      setBusy(false);
   518→    }
   519→  };
   520→  return (
   521→    <FcCard className="p-6 sm:p-8">
   522→      <BackButton onClick={() => goto({ k: "welcome" })} />
   523→      <h2 className="fc-serif mb-1 text-center text-2xl font-bold text-[#1E4D2B]">Code PIN</h2>
   524→      <p className="mb-6 text-center text-[14px] text-[#5C6B60]">
   525→        Votre PIN de secours, {state.name.split(" ")[0]}.
   526→      </p>
   527→      <FcInput
   528→        icon={<Lock className="h-5 w-5" />}
   529→        placeholder="● ● ● ●"
   530→        value={pin}
   531→        onChange={(e) => setPin(e.target.value.replace(/\D/g, "").slice(0, 6))}
   532→        onKeyDown={(e) => e.key === "Enter" && pin.length >= 4 && handle()}
   533→        inputMode="numeric"
   534→        type="password"
   535→        maxLength={6}
   536→        className="mb-4 text-center tracking-[0.4em]"
   537→        aria-label="Code PIN"
   538→      />
   539→      <FcButton onClick={handle} disabled={busy || pin.length < 4}>
   540→        {busy ? <FcSpinner /> : "Se connecter"}
   541→      </FcButton>
   542→      <p className="mt-4 text-center text-[13px] text-[#8A9288]">
   543→        5 tentatives maximum, puis verrouillage de 30 minutes.
   544→      </p>
   545→    </FcCard>
   546→  );
   547→}

   549→/* ───────────────────────── Écran de premières étapes (post-activation) ───────────────────────── */
   550→
   551→function SetupScreen({
   552→  state,
   553→  busy,
   554→  setBusy,
   555→  onReady,
   556→}: {
   557→  state: Extract<Step, { k: "setup" }>;
   558→  busy: boolean;
   559→  setBusy: (b: boolean) => void;
   560→  onReady: (name: string) => void;
   561→}) {
   562→  const [ack, setAck] = useState(false);
   563→  const [copied, setCopied] = useState(false);
   564→  const [pin, setPin] = useState("");
   565→  const [pinDone, setPinDone] = useState(false);
   566→  const [bioDone, setBioDone] = useState(false);
   567→  const [bioSimulated, setBioSimulated] = useState(false);
   568→
   569→  const copyRecovery = async () => {
   570→    await navigator.clipboard.writeText(state.recoveryCode).catch(() => {});
   571→    setCopied(true);
   572→    toast.success("Code de récupération copié — collez-le dans un endroit sûr.");
   573→  };
   574→
   575→  const savePin = async () => {
   576→    setBusy(true);
   577→    try {
   578→      const res = await api("/api/auth/pin", { body: { action: "setup", pin } });
   579→      if (!res.ok) throw new Error(res.error);
   580→      setPinDone(true);
   581→      toast.success("Code PIN enregistré ✓");
   582→    } catch (e) {
   583→      toast.error(e instanceof Error ? e.message : "Échec de l'enregistrement du PIN.");
   584→    } finally {
   585→      setBusy(false);
   586→    }
   587→  };
   588→
   589→  const enrollBiometry = async () => {
   590→    setBusy(true);
   591→    try {
   592→      const { options } = await api<{ options: unknown }>("/api/auth/passkey", {
   593→        body: { action: "register-options" },
   594→      });
   595→      const response = await startRegistration({ optionsJSON: options as never });
   596→      const result = await api("/api/auth/passkey", {
   597→        body: { action: "register-verify", response, deviceLabel: deviceName() },
   598→      });
   599→      if (!result.ok) throw new Error(result.error);
   600→      setBioDone(true);
   601→      localStorage.setItem("fc_has_passkey", "1");
   602→      toast.success("Biométrie activée ✓ Vous vous connecterez d'un simple geste.");
   603→    } catch (e) {
   604→      toast.error(
   605→        e instanceof Error
   606→          ? `Biométrie indisponible ici (${e.message.slice(0, 60)}). Utilisez le mode démo ci-dessous.`
   607→          : "Biométrie indisponible."
   608→      );
   609→    } finally {
   610→      setBusy(false);
   611→    }
   612→  };
   613→
   614→  const simulateBiometry = async () => {
   615→    setBusy(true);
   616→    try {
   617→      const result = await api("/api/auth/passkey", { body: { action: "simulate-register" } });
   618→      if (!result.ok) throw new Error(result.error);
   619→      setBioDone(true);
   620→      setBioSimulated(true);
   621→      localStorage.setItem("fc_has_passkey", "0");
   622→      toast.success("Appareil biométrique simulé enregistré (mode démo) ✓");
   623→    } catch (e) {
   624→      toast.error(e instanceof Error ? e.message : "Simulation impossible.");
   625→    } finally {
   626→      setBusy(false);
   627→    }
   628→  };
   629→
   630→  const enter = () => {
   631→    if (!ack) {
   632→      toast.error("Confirmez d'abord avoir noté votre code de récupération.");
   633→      return;
   634→    }
   635→    onReady(state.userName);
   636→  };
   637→
   638→  return (
   639→    <div className="space-y-4">
   640→      {/* 1. Code de récupération */}
   641→      <FcCard className="p-6">
   642→        <div className="mb-2 flex items-center gap-2">
   643→          <LifeBuoy className="h-5 w-5 text-[#D97706]" />
   644→          <h2 className="fc-serif text-xl font-bold text-[#1E4D2B]">Votre code de récupération</h2>
   645→        </div>
   646→        <p className="mb-3 text-[13px] leading-relaxed text-[#5C6B60]">
   647→          {state.reenroll
   648→            ? "Un NOUVEAU code vous a été attribué (l'ancien est invalidé)."
   649→            : "C'est la seule fois où il sera affiché."}{" "}
   650→          Notez-le sur papier ou envoyez-le-vous par WhatsApp. Il permet de récupérer votre accès si vous perdez
   651→          votre téléphone.
   652→        </p>
   653→        <div className="mb-3 flex items-center justify-center gap-3 rounded-2xl border-2 border-dashed border-[#D97706] bg-[#D97706]/5 p-4">
   654→          <span className="fc-serif text-2xl font-bold tracking-wider text-[#7A5B2E]">{state.recoveryCode}</span>
   655→          <button
   656→            onClick={copyRecovery}
   657→            className="rounded-lg p-2 text-[#C89F62] hover:bg-[#C89F62]/10"
   658→            aria-label="Copier le code de récupération"
   659→          >
   660→            {copied ? <Check className="h-5 w-5" /> : <Copy className="h-5 w-5" />}
   661→          </button>
   662→        </div>
   663→        <label className="flex cursor-pointer items-start gap-2 text-[14px] text-[#2B3530]">
   664→          <input
   665→            type="checkbox"
   666→            checked={ack}
   667→            onChange={(e) => setAck(e.target.checked)}
   668→            className="mt-0.5 h-4 w-4 accent-[#1F5B3F]"
   669→          />
   670→          <span>J&apos;ai noté mon code de récupération en lieu sûr.</span>
   671→        </label>
   672→      </FcCard>
   673→
   674→      {/* 2. PIN de secours */}
   675→      <FcCard className="p-6">
   676→        <div className="mb-2 flex items-center gap-2">
   677→          <Lock className="h-5 w-5 text-[#2A4A3A]" />
   678→          <h2 className="fc-serif text-xl font-bold text-[#1E4D2B]">Code PIN de secours</h2>
   679→        </div>
   680→        {pinDone ? (
   681→          <p className="text-sm font-medium text-[#16A34A]">✓ PIN défini — il déverrouille cet appareil.</p>
   682→        ) : (
   683→          <>
   684→            <p className="mb-3 text-[13px] text-[#5C6B60]">
   685→              4 à 6 chiffres, mémorisable. Sert quand la biométrie est indisponible. (Recommandé, mais optionnel.)
   686→            </p>
   687→            <div className="flex gap-2">
   688→              <FcInput
   689→                placeholder="● ● ● ●"
   690→                value={pin}
   691→                onChange={(e) => setPin(e.target.value.replace(/\D/g, "").slice(0, 6))}
   692→                inputMode="numeric"
   693→                type="password"
   694→                maxLength={6}
   695→                className="flex-1"
   696→                aria-label="Nouveau code PIN"
   697→              />
   698→              <FcButton onClick={savePin} disabled={busy || pin.length < 4} className="w-auto px-6">
   699→                Définir
   700→              </FcButton>
   701→            </div>
   702→          </>
   703→        )}
   704→      </FcCard>
   705→
   706→      {/* 3. Biométrie */}
   707→      <FcCard className="p-6">
   708→        <div className="mb-2 flex items-center gap-2">
   709→          <Fingerprint className="h-5 w-5 text-[#2A4A3A]" />
   710→          <h2 className="fc-serif text-xl font-bold text-[#1E4D2B]">Connexion par empreinte</h2>
   711→        </div>
   712→        {bioDone ? (
   713→          <p className="text-sm font-medium text-[#16A34A]">
   714→            ✓ {bioSimulated ? "Appareil simulé enregistré (démo)" : "Biométrie active sur cet appareil"}
   715→          </p>
   716→        ) : (
   717→          <div className="space-y-2">
   718→            <p className="mb-2 text-[13px] text-[#5C6B60]">
   719→              Un geste = connecté, pour toujours. Zéro mot de passe à retenir.
   720→            </p>
   721→            <FcButton onClick={enrollBiometry} disabled={busy}>
   722→              {busy ? <FcSpinner /> : <Fingerprint className="h-5 w-5" />} Activer mon empreinte
   723→            </FcButton>
   724→            <FcButton variant="outline" onClick={simulateBiometry} disabled={busy}>
   725→              <ShieldCheck className="h-4 w-4" /> Simuler sur cet appareil (démo)
   726→            </FcButton>
   727→          </div>
   728→        )}
   729→      </FcCard>
   730→
   731→      <FcButton onClick={enter} className="h-[54px] text-[17px]">
   732→        Entrer dans FamilleConnect
   733→      </FcButton>
   734→    </div>
   735→  );
   736→}
