   296→/* ───────────────────────── Saisie du numéro (inscription & connexion) ───────────────────────── */
   297→
   298→function PhoneScreen({
   299→  mode,
   300→  busy,
   301→  setBusy,
   302→  goto,
   303→}: {
   304→  mode: "signup" | "login";
   305→  busy: boolean;
   306→  setBusy: (b: boolean) => void;
   307→  goto: (s: Step) => void;
   308→}) {
   309→  const [digits, setDigits] = useState("");
   310→  const { country, setCountry } = usePhoneCountry();
   311→  const maxD = maxDigitsFor(country);
   312→
   313→  const send = async () => {
   314→    if (!phoneDigitsOk(country, digits)) {
   315→      toast.error(
   316→        country.iso === "CM"
   317→          ? "Entrez un numéro camerounais à 9 chiffres (ex : 6 91 23 45 67)."
   318→          : `Entrez votre numéro ${country.fr} sans l'indicatif +${country.dial}${country.len ? ` (${country.len} chiffres)` : " (6 à 14 chiffres)"}.`
   319→      );
   320→      return;
   321→    }
   322→    setBusy(true);
   323→    try {
   324→      const res = await api<{ verifyToken: string; demoCode: string; known: boolean }>("/api/auth/phone", {
   325→        body: { action: "request", phone: country.dial + digits, country: country.iso, deviceName: deviceName() },
   326→      });
   327→      if (!res.ok) throw new Error(res.error);
   328→      try {
   329→        localStorage.setItem("fc_phone", "+" + country.dial + digits);
   330→      } catch {
   331→        /* rien */
   332→      }
   333→      toast.success(`Notification de vérification envoyée au +${country.dial} ${groupPhone(digits)}`, {
   334→        description: "Consultez vos notifications — l'application vérifie ensuite votre appareil.",
   335→      });
   336→      goto({ k: "verify", verifyToken: res.verifyToken, digits, demoCode: res.demoCode, dial: country.dial, iso: country.iso });
   337→    } catch (e) {
   338→      toast.error(e instanceof Error ? e.message : "Envoi de la notification impossible.");
   339→    } finally {
   340→      setBusy(false);
   341→    }
   342→  };
   343→
   344→  return (
   345→    <FcCard className="p-6 sm:p-8">
   346→      <BackButton onClick={() => goto(mode === "signup" ? { k: "welcome" } : { k: "login-choice" })} />
   347→      <StepBadge current={1} total={mode === "signup" ? 3 : 2} label={mode === "signup" ? "Inscription" : "Connexion"} />
   348→      <h2 className="fc-serif mb-1 text-center text-2xl font-bold text-[#1E4D2B]">Votre numéro de téléphone</h2>
   349→      <p className="mb-6 text-center text-[14px] leading-relaxed text-[#5C6B60]">
   350→        {mode === "signup"
   351→          ? "Votre numéro est votre identifiant — nous vous envoyons une notification pour vérifier qu'il s'agit bien de votre téléphone."
   352→          : "Nous vous envoyons une notification avec un code pour vous reconnecter sans mot de passe."}
   353→      </p>
   354→
   355→      <div className="mb-4 flex items-stretch gap-2">
   356→        <PhoneCountrySelect
   357→          value={country}
   358→          onChange={(c) => {
   359→            setCountry(c);
   360→            setDigits(""); // nouveau pays → nouveau numéro
   361→          }}
   362→        />
   363→        <FcInput
   364→          className="flex-1"
   365→          placeholder={placeholderFor(country)}
   366→          value={groupPhone(digits)}
   367→          onChange={(e) => setDigits(onlyDigits(e.target.value).slice(0, maxD))}
   368→          onKeyDown={(e) => e.key === "Enter" && phoneDigitsOk(country, digits) && send()}
   369→          inputMode="tel"
   370→          autoComplete="tel-national"
   371→          maxLength={20}
   372→          aria-label="Numéro de téléphone (sans indicatif)"
   373→        />
   374→      </div>
   375→
   376→      <FcButton onClick={send} disabled={busy || !phoneDigitsOk(country, digits)} className="h-[54px]">
   377→        {busy ? <FcSpinner /> : <BellRing className="h-5 w-5" />}
   378→        Recevoir la notification
   379→      </FcButton>
   380→
   381→      <p className="mt-4 text-center text-[12px] text-[#8A9288]">
   382→        {country.iso === "CM"
   383→          ? "Format camerounais — 9 chiffres commençant par 6 (mobile) ou 2 (fixe)."
   384→          : `L'indicatif +${country.dial} est ajouté automatiquement — saisissez le numéro sans l'indicatif.`}
   385→      </p>
   386→    </FcCard>
   387→  );
   388→}

   390→/* ───────────────────────── Vérification du numéro (OTP) ───────────────────────── */
   391→
   392→function VerifyScreen({
   393→  state,
   394→  busy,
   395→  setBusy,
   396→  goto,
   397→  onReady,
   398→}: {
   399→  state: Extract<Step, { k: "verify" }>;
   400→  busy: boolean;
   401→  setBusy: (b: boolean) => void;
   402→  goto: (s: Step) => void;
   403→  onReady: (name: string) => void;
   404→}) {
   405→  const [code, setCode] = useState("");
   406→  const [manual, setManual] = useState(false);
   407→  const [phase, setPhase] = useState<"sent" | "reading" | "detected">("sent");
   408→  const verifiedRef = useRef(false);
   409→
   410→  const doVerify = useCallback(
   411→    async (c: string) => {
   412→      if (verifiedRef.current) return;
   413→      verifiedRef.current = true;
   414→      setBusy(true);
   415→      try {
   416→        const res = await api<{
   417→          registered: boolean;
   418→          user?: { id: string; fullName: string };
   419→          registerToken?: string;
   420→        }>("/api/auth/phone", {
   421→          body: { action: "verify", verifyToken: state.verifyToken, code: c, deviceName: deviceName() },
   422→        });
   423→        if (!res.ok) {
   424→          verifiedRef.current = false;
   425→          throw new Error(res.error);
   426→        }
   427→        if (res.registered && res.user) {
   428→          // Le numéro est déjà enregistré → session ouverte, entrée directe
   429→          toast.success(`Content de vous revoir, ${res.user.fullName.split(" ")[0]} !`);
   430→          void afterAuthPush(false);
   431→          onReady(res.user.fullName);
   432→        } else if (res.registerToken) {
   433→          toast.info("Numéro vérifié — complétez votre profil pour créer votre compte.");
   434→          goto({ k: "profile", registerToken: res.registerToken, digits: state.digits, dial: state.dial, iso: state.iso });
   435→        } else {
   436→          throw new Error("Réponse inattendue du serveur.");
   437→        }
   438→      } catch (e) {
   439→        verifiedRef.current = false;
   440→        setManual(true);
   441→        setCode("");
   442→        toast.error(e instanceof Error ? e.message : "Vérification impossible.");
   443→      } finally {
   444→        setBusy(false);
   445→      }
   446→    },
   447→    [state.verifyToken, state.digits, setBusy, goto, onReady]
   448→  );
   449→
   450→  // Séquence automatique : la notification arrive, l'application lit le code
   451→  // sur le téléphone (comme WhatsApp lit le SMS) puis vérifie toute seule.
   452→  useEffect(() => {
