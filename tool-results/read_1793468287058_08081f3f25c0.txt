   131→/** Jour du 2ᵉ dimanche du mois (8 à 14). */
   132→export function secondSunday(year: number, month0: number): number {
   133→  const dow = new Date(year, month0, 1).getDay(); // 0 = dimanche
   134→  return 1 + ((7 - dow) % 7) + 7;
   135→}

   137→export function isoDate(d: Date): string {
   138→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;
   139→}

   159→/** Construit les 12 mois de l'année : 11 tours + l'AG de décembre. */
   160→export function buildAnnee(year: number, roster: Member[], today = new Date()): TourMois[] {
   161→  const byTour = new Map<number, Member>();
   162→  roster.forEach((m) => byTour.set(m.tour, m));
   163→  const moisCourant = today.getMonth();
   164→  return MOIS_FR.map((mois, m0) => {
   165→    const jour = secondSunday(year, m0);
   166→    const date = new Date(year, m0, jour);
   167→    const ag = m0 === 11;
   168→    const beneficiary = ag ? null : byTour.get(m0 + 1) ?? null;
   169→    const pot = ag ? 0 : potMensuel(roster.length);
   170→    const collation = ag ? 0 : collationMensuelle(roster.length);
   171→    const etat: EtatMois = m0 < moisCourant ? "passe" : m0 === moisCourant ? "courant" : "futur";
   172→    return {
   173→      month0: m0, mois, annee: year, jour, date,
   174→      iso: `${year}-${String(m0 + 1).padStart(2, "0")}-${String(jour).padStart(2, "0")}`,
   175→      tour: ag ? null : m0 + 1, beneficiary, ag, pot, collation, total: pot + collation, etat,
   176→    };
   177→  });
   178→}

   180→/** État de la réunion d'un mois au niveau du jour : passée / aujourd'hui / à venir. */
   181→export function etatReunion(t: TourMois, today = new Date()): "passee" | "aujourdhui" | "avenir" {
   182→  const t0 = new Date(today.getFullYear(), today.getMonth(), today.getDate()).getTime();
   183→  const d0 = new Date(t.annee, t.month0, t.jour).getTime();
   184→  if (d0 < t0) return "passee";
   185→  if (d0 === t0) return "aujourdhui";
   186→  return "avenir";
   187→}

   189→/** Nombre de jours avant la réunion (négatif = passée). */
   190→export function joursAvant(t: TourMois, today = new Date()): number {
   191→  const t0 = new Date(today.getFullYear(), today.getMonth(), today.getDate()).getTime();
   192→  const d0 = new Date(t.annee, t.month0, t.jour).getTime();
   193→  return Math.round((d0 - t0) / 86_400_000);
   194→}

   261→    if (t.ag) return { tour: t, member: null, cotis: 0, colla: 0, total: 0, statut: "ag" as StatutVersement, paidAt: null };
   262→    const b = t.beneficiary!;
   263→    if (b.id === vousId) {
   264→      return { tour: t, member: b, cotis: 0, colla: TONTINE.montantCollation, total: TONTINE.montantCollation, statut: "mon_tour" as StatutVersement, paidAt: null };
   265→    }
   266→    const paidAt = payes[payeKey(b.id, t.annee, t.month0)] ?? null;
   267→    if (paidAt) return { tour: t, member: b, cotis: TONTINE.montantCotisation, colla: TONTINE.montantCollation, total: detteParPersonne(), statut: "paye" as StatutVersement, paidAt };
   268→    if (t.iso <= todayIso) return { tour: t, member: b, cotis: TONTINE.montantCotisation, colla: TONTINE.montantCollation, total: detteParPersonne(), statut: "a_payer" as StatutVersement, paidAt: null };
   269→    return { tour: t, member: b, cotis: TONTINE.montantCotisation, colla: TONTINE.montantCollation, total: detteParPersonne(), statut: "avenir" as StatutVersement, paidAt: null };
   270→  });
   271→}

   273→/** Statut de cotisation d'un membre pour la réunion du mois courant. */
   274→export function statutMembreDuMois(
   275→  m: Member,
   276→  moisCourant: TourMois | undefined,
   277→  payes: Payes,
   278→  today = new Date(),
   279→): { label: string; cls: string } {
   280→  const neutre = "bg-[#2A4A3A]/8 text-[#5C6B60] border-[#E5DCC0]";
   281→  if (!moisCourant) return { label: "—", cls: neutre };
   282→  if (moisCourant.ag) return { label: `AG le ${moisCourant.jour}`, cls: "bg-[#3D6B72]/10 text-[#3D6B72] border-[#3D6B72]/30" };
   283→  if (moisCourant.beneficiary?.id === m.id) {
   284→    return etatReunion(moisCourant, today) === "passee"
   285→      ? { label: "A reçu la réunion ✓", cls: "bg-[#2F6B45]/12 text-[#2F6B45] border-[#2F6B45]/30" }
   286→      : { label: "Son tour — reçoit la réunion", cls: "bg-[#C89F62]/18 text-[#8A6420] border-[#C89F62]/45" };
   287→  }
   288→  if (etatReunion(moisCourant, today) === "avenir") {
   289→    return { label: `Réunion le ${moisCourant.jour}`, cls: neutre };
   290→  }
   291→  if (m.isVous) {
   292→    const paid = !!payes[payeKey(m.id, moisCourant.annee, moisCourant.month0)];
   293→    return paid
   294→      ? { label: "Payé ✓", cls: "bg-[#2F6B45]/12 text-[#2F6B45] border-[#2F6B45]/30" }
   295→      : { label: "En attente", cls: "bg-[#A5473D]/10 text-[#A5473D] border-[#A5473D]/30" };
   296→  }
   297→  // Membres fictifs — registre du trésorier : déterministe (≈ 1 sur 4 en attente).
   298→  const paye = (m.tour + moisCourant.month0) % 4 !== 0;
   299→  return paye
   300→    ? { label: "Payé ✓", cls: "bg-[#2F6B45]/12 text-[#2F6B45] border-[#2F6B45]/30" }
   301→    : { label: "En attente", cls: "bg-[#C89F62]/18 text-[#8A6420] border-[#C89F62]/45" };

   329→/** Journal de caisse de l'année — soldes dérivés des règles et des anomalies. */
   330→export function journalDeCaisse(annee: TourMois[], roster: Member[], today = new Date()): JournalRow[] {
   331→  const n = roster.length;
   332→  let caisse = n * TONTINE.fraisAdhesion; // adhésions encaissées en janvier
   333→  return annee.map((t) => {
   334→    const penalites = t.etat === "futur" ? 0 : ANOMALIES.filter((a) => a.month0 === t.month0).reduce((s, a) => s + a.montant, 0);
   335→    if (t.ag) {
   336→      return { tour: t, encaisse: 0, pot: 0, collation: 0, caisseSociale: caisse, penalites: 0, etat: t.etat === "futur" ? ("avenir" as const) : ("cloture" as const), note: "Assemblée générale — aucun mouvement" };
   337→    }
   338→    if (t.etat === "futur") {
   339→      return { tour: t, encaisse: t.total, pot: t.pot, collation: t.collation, caisseSociale: caisse, penalites: 0, etat: "avenir" as const, note: "Projection" };
   340→    }
   341→    caisse += penalites;
   342→    return {
   343→      tour: t, encaisse: t.total, pot: t.pot, collation: t.collation, caisseSociale: caisse, penalites,
   344→      etat: t.etat === "courant" ? ("en_cours" as const) : ("cloture" as const),
   345→      note: penalites > 0 ? `+ ${penalites.toLocaleString("fr-FR")} de pénalités` : "",
   346→    };
   347→  });
   348→}

   360→/** Classement des cotisants — les fictifs selon le registre, « Vous » selon votre suivi. */
   361→export function classementCotisants(annee: TourMois[], roster: Member[], payes: Payes, vousId: string, today = new Date()): ClassementRow[] {
   362→  const todayIso = isoDate(today);
   363→  const toursEcoules = annee.filter((t) => !t.ag && t.iso <= todayIso);
   364→  return roster.map((m) => {
   365→    const mes = toursEcoules.filter((t) => t.beneficiary?.id !== m.id);
   366→    const parts = mes.length;
   367→    const anoms = ANOMALIES.filter((a) => a.memberId === m.id && mes.some((t) => t.month0 === a.month0));
   368→    const versees = m.isVous
   369→      ? mes.filter((t) => payes[payeKey(m.id, t.annee, t.month0)]).length
   370→      : parts; // registre officiel : versé, même avec pénalité de retard
   371→    const ponctualite = parts > 0 ? Math.round((versees / parts) * 100) : null;
   372→    return {
   373→      member: m, parts, versees,
   374→      retards: anoms.length,
   375→      penalites: anoms.reduce((s, a) => s + a.montant, 0),
   376→      ponctualite,
   377→      totalVerse: versees * detteParPersonne(),
   378→    };
   379→  }).sort((a, b) => {
   380→    const pa = a.ponctualite ?? -1;
   381→    const pb = b.ponctualite ?? -1;
   382→    if (pb !== pa) return pb - pa;
   383→    if (b.totalVerse !== a.totalVerse) return b.totalVerse - a.totalVerse;
   384→    return a.member.tour - b.member.tour;
   385→  });
   386→}
   387→
   388→/** Projection de fin d'année — tous les montants dérivés des règles. */
   389→export function projectionAnnee(annee: TourMois[], roster: Member[]) {
   390→  const tours = annee.filter((t) => !t.ag);
   391→  const nbTours = tours.length;
   392→  const pot = potMensuel(roster.length);
   393→  const colla = collationMensuelle(roster.length);
   394→  const penalites = ANOMALIES.reduce((s, a) => s + a.montant, 0);
   395→  return {
   396→    nbTours,
   397→    totalCollecte: nbTours * (pot + colla),
   398→    totalPots: nbTours * pot,
   399→    totalCollations: nbTours * colla,
   400→    caisseSociale: roster.length * TONTINE.fraisAdhesion + penalites,
   401→    penalites,
   402→  };
   403→}

   405→/** Caisses à l'instant présent (tours déjà arrivés). */
   406→export function caissesDuMoment(annee: TourMois[], roster: Member[], today = new Date()) {
   407→  const todayIso = isoDate(today);
   408→  const arrivés = annee.filter((t) => !t.ag && t.iso <= todayIso);
   409→  const penalites = ANOMALIES.filter((a) => arrivés.some((t) => t.month0 === a.month0)).reduce((s, a) => s + a.montant, 0);
   410→  return {
   411→    collecte: arrivés.reduce((s, t) => s + t.total, 0),
   412→    potsVerses: arrivés.reduce((s, t) => s + t.pot, 0),
   413→    collations: arrivés.reduce((s, t) => s + t.collation, 0),
   414→    caisseSociale: roster.length * TONTINE.fraisAdhesion + penalites,
   415→  };
   416→}

   457→
   458→   ÉPARGNE — intérêts SIMPLES : chaque réunion tenue après le dépôt ajoute
   459→   capital × taux, sur la base du capital uniquement (jamais sur les intérêts).
   460→   10 000 épargnés en mars à 3 % : avril 10 300 · mai 10 600 · juin 10 900 …
   461→   jusqu'en décembre de l'année du dépôt, où le tout est récupéré.
   462→
   463→   PRÊT — argent emprunté à la réunion : les intérêts courent au taux du mois
   464→   pendant 3 MOIS (10 000 empruntés en février → 10 900 après 3 mois). Passée
   465→   l'échéance sans remboursement, la dette est recalculée chaque mois à taux
   466→   composé SUR LA NOUVELLE BASE (3 % de 10 900 le mois suivant, etc.). */

   468→export const TAUX_DEFAUT = 3; // % par mois — valeur par défaut, paramétrable par le bureau
   469→

   472→const TAUX_KEY = "fc_tontine_taux_v1";
   473→
   474→/** Taux mensuel partagé (épargnes et prêts), en % — borné 0 à 20. */
   475→export function loadTaux(): number {
   476→  try {
   477→    const raw = localStorage.getItem(TAUX_KEY);
   478→    if (!raw || raw.trim() === "") return TAUX_DEFAUT;
   479→    const v = Number(raw);
   480→    return isFinite(v) && v >= 0 && v <= 20 ? Math.round(v * 10) / 10 : TAUX_DEFAUT;
   481→  } catch {
   482→    return TAUX_DEFAUT;
   483→  }
   484→}
   485→
   486→export function saveTaux(t: number): void {
   487→  try {
   488→    localStorage.setItem(TAUX_KEY, String(Math.round(t * 10) / 10));
   489→  } catch {
   490→    /* stockage indisponible — réglage gardé en mémoire pour la session */
   491→  }
   492→}

   494→/** Intérêt d'UN mois d'épargne : capital × taux (base constante — non cumulatif). */
   495→export const interetEpargneMois = (capital: number, tauxPct: number) => Math.round(capital * (tauxPct / 100));
   496→
   497→/** Solde d'une épargne après n mois comptés : capital + n × intérêt mensuel. */
   498→export const soldeEpargneAuMois = (capital: number, tauxPct: number, n: number) =>
   499→  n <= 0 ? capital : capital + interetEpargneMois(capital, tauxPct) * n;
   500→
   501→/** Dette d'un prêt au mois `cible` (hors remboursement) : 3 mois d'intérêts
   502→ *  simples sur le capital, puis capitalisation mensuelle sur la nouvelle base. */
   503→export function dettePretAuMois(capital: number, tauxPct: number, depart: number, cible: number): number {
   504→  if (cible <= depart) return capital;
   505→  const r = tauxPct / 100;
   506→  const parMois = Math.round(capital * r);
   507→  let solde = capital;
   508→  for (let m = depart + 1; m <= Math.min(cible, 11); m++) {
   509→    solde += m - depart <= 3 ? parMois : Math.round(solde * r);
   510→  }
   511→  return solde;
   512→}

   514→/** Nombre de réunions déjà tenues APRÈS le mois de départ (l'épargne ou le
   515→ *  prêt commence à courir le mois suivant) — plafonné à décembre. */
   516→export function moisEcoules(depart: number, annee: TourMois[], today = new Date()): number {
   517→  const todayIso = isoDate(today);
   518→  let n = 0;
   519→  for (let m = depart + 1; m <= 11; m++) {
   520→    if (annee[m] && annee[m].iso <= todayIso) n++;
   521→  }
   522→  return n;
   523→}
