   389→function NotifTab({ session }: { session: SessionData }) {
   390→  const [items, setItems] = useState<NotificationItem[]>([]);
   391→  const [loading, setLoading] = useState(true);
   392→  const [prefs, setPrefs] = useState<{ digestEnabled: boolean; quietStart: number; quietEnd: number; pushEnabled: boolean } | null>(null);
   393→
   394→  const load = useCallback(async () => {
   395→    setLoading(true);
   396→    const res = await api<{ notifications: NotificationItem[]; prefs: typeof prefs }>("/api/notifications");
   397→    if (res.ok) {
   398→      setItems(res.notifications);
   399→      setPrefs(res.prefs);
   400→    }
   401→    setLoading(false);
   402→  }, []);
   403→
   404→  useEffect(() => {
   405→    const t = setTimeout(load, 0);
   406→    return () => clearTimeout(t);
   407→  }, [load]);
   408→
   409→  const markRead = async (id: string) => {
   410→    await api("/api/notifications", { body: { action: "read", id } });
   411→    setItems((prev) => prev.map((n) => (n.id === id ? { ...n, read: true } : n)));
   412→  };
   413→
   414→  const markAll = async () => {
   415→    const res = await api<{ readAll: number }>("/api/notifications", { body: { action: "read-all" } });
   416→    if (res.ok) {
   417→      setItems((prev) => prev.map((n) => ({ ...n, read: true })));
   418→      toast.success(`${res.readAll} notification(s) marquée(s) comme lue(s)`);
   419→    }
   420→  };
   421→
   422→  const toggleDigest = async () => {
   423→    if (!prefs) return;
   424→    const res = await api<{ prefs: typeof prefs }>("/api/notifications", {
   425→      body: { action: "prefs", digestEnabled: !prefs.digestEnabled },
