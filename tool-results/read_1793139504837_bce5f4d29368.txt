   125→function HomeTab({ session, goTab }: { session: SessionData; goTab: (t: "notif" | "security") => void }) {
   126→  const [pushState, setPushState] = useState<"unknown" | "unsupported" | "default" | "granted" | "denied" | "subscribed">("unknown");
   127→  const [busyPush, setBusyPush] = useState(false);
   128→
   129→  useEffect(() => {
   130→    if (typeof Notification === "undefined" || !("serviceWorker" in navigator)) {
   131→      const t = setTimeout(() => setPushState("unsupported"), 0);
   132→      return () => clearTimeout(t);
   133→    }
   134→    const t = setTimeout(() => {
   135→      setPushState(Notification.permission === "granted" ? "granted" : Notification.permission === "denied" ? "denied" : "default");
   136→    }, 0);
   137→    return () => clearTimeout(t);
   138→  }, []);
   139→
   140→  const enablePush = useCallback(async () => {
   141→    setBusyPush(true);
   142→    try {
   143→      const perm = await Notification.requestPermission();
   144→      setPushState(perm === "granted" ? "granted" : perm === "denied" ? "denied" : "default");
   145→      if (perm !== "granted") {
   146→        toast.warning("Permission refusée — vous pouvez l'accorder plus tard dans les réglages du navigateur.");
   147→        return;
   148→      }
   149→      const reg = await navigator.serviceWorker.register("/sw.js");
   150→      await navigator.serviceWorker.ready;
   151→      const vapid = await api<{ publicKey: string }>("/api/push");
   152→      if (!vapid.ok) throw new Error(vapid.error);
   153→      const sub = await reg.pushManager.subscribe({
   154→        userVisibleOnly: true,
   155→        applicationServerKey: urlBase64ToUint8Array(vapid.publicKey),
   156→      });
   157→      const saved = await api("/api/push", { body: { action: "subscribe", subscription: sub.toJSON() } });
   158→      if (!saved.ok) throw new Error(saved.error);
   159→      setPushState("subscribed");
   160→      toast.success("Notifications activées ✓ Vous recevrez les alertes de la famille même app fermée.");
   161→    } catch (e) {
   162→      toast.error(e instanceof Error ? e.message : "Activation des notifications impossible.");
   163→    } finally {
   164→      setBusyPush(false);
   165→    }
   166→  }, []);
   167→
   168→  const testPush = useCallback(async () => {
   169→    setBusyPush(true);
   170→    try {
   171→      const res = await api("/api/push", { body: { action: "test" } });
   172→      if (!res.ok) throw new Error(res.error);
   173→      toast.success("Push envoyé ! Regardez votre écran — même vérrouillé.");
   174→    } catch (e) {
   175→      toast.error(e instanceof Error ? e.message : "Envoi impossible.");
   176→    } finally {
   177→      setBusyPush(false);
   178→    }
   179→  }, []);
   180→
   181→  return (
   182→    <div className="space-y-4">
   183→      {/* Carte de bienvenue */}
   184→      <FcCard className="overflow-hidden">
   185→        <div className="bg-[#1E4D2B] px-5 py-4 text-white">
   186→          <p className="text-[13px] opacity-80">Karibu,</p>
   187→          <p className="fc-serif text-xl font-bold">{session.user.fullName}</p>
   188→        </div>
   189→        <div className="space-y-2 px-5 py-4 text-[14px] text-[#2B3530]">
   190→          {session.member && (
   191→            <>
   192→              <div className="flex justify-between"><span className="text-[#8A9288]">Branche</span><span className="font-semibold">{session.member.branch}</span></div>
   193→              <div className="flex justify-between"><span className="text-[#8A9288]">Rôle</span><span className="font-semibold">{roleLabel(session.member.role)}</span></div>
   194→              <div className="flex justify-between"><span className="text-[#8A9288]">Statut</span><span className="font-semibold text-[#16A34A]">Compte actif</span></div>
   195→            </>
   196→          )}
   197→          <div className="flex justify-between"><span className="text-[#8A9288]">Appareil</span><span className="font-semibold">{session.device?.name}</span></div>
   198→        </div>
   199→      </FcCard>
   200→
   201→      {/* Carte notifications push */}
   202→      <FcCard className="p-5">
   203→        <div className="mb-2 flex items-center gap-2">
   204→          <BellRing className="h-5 w-5 text-[#C89F62]" />
   205→          <h3 className="fc-serif text-lg font-bold text-[#1E4D2B]">Notifications push</h3>
   206→        </div>
   207→        {pushState === "unsupported" && (
   208→          <p className="text-[13px] text-[#8A9288]">
   209→            Votre navigateur actuel ne les prend pas en charge. Sur téléphone : installez l&apos;app via
   210→            « Ajouter à l&apos;écran d&apos;accueil » puis rouvrez-la.
   211→          </p>
   212→        )}
   213→        {pushState === "default" && (
   214→          <>
   215→            <p className="mb-3 text-[13px] text-[#5C6B60]">
   216→              Recevez les événements de la famille même application fermée — comme WhatsApp. Gratuit, illimité.
   217→            </p>
   218→            <FcButton onClick={enablePush} disabled={busyPush}>
   219→              {busyPush ? "Activation…" : <><BellRing className="h-5 w-5" /> Activer les notifications</>}
   220→            </FcButton>
   221→          </>
   222→        )}
   223→        {(pushState === "granted" || pushState === "subscribed") && (
   224→          <>
   225→            <p className="mb-3 text-[13px] text-[#16A34A]">
   226→              ✓ {pushState === "subscribed" ? "Abonnement push actif sur cet appareil." : "Permission accordée."}
   227→            </p>
   228→            <FcButton variant="gold" onClick={testPush} disabled={busyPush}>
   229→              {busyPush ? "Envoi…" : <><Send className="h-4 w-4" /> Tester la notification réelle</>}
   230→            </FcButton>
   231→          </>
   232→        )}
   233→        {pushState === "denied" && (
   234→          <p className="flex items-center gap-2 text-[13px] text-[#B91C1C]">
   235→            <BellOff className="h-4 w-4" /> Notifications bloquées — réautorisez-les dans les réglages du site.
   236→          </p>
   237→        )}
   238→      </FcCard>
   239→
   240→      {/* Raccourcis */}
   241→      <div className="grid grid-cols-2 gap-3">
   242→        <ShortcutCard icon={<Bell className="h-5 w-5" />} label="Centre d'alertes" onClick={() => goTab("notif")} />
   243→        <ShortcutCard icon={<ShieldCheck className="h-5 w-5" />} label="Ma sécurité" onClick={() => goTab("security")} />
   244→      </div>
   245→
   246→      <FcCard className="p-5">
   247→        <div className="flex items-start gap-3">
   248→          <TreePine className="mt-0.5 h-5 w-5 shrink-0 text-[#3D6B52]" />
   249→          <p className="text-[13px] leading-relaxed text-[#5C6B60]">
   250→            <strong className="text-[#1E4D2B]">Prochaines étapes du MVP :</strong> registre complet des membres (U1),
   251→            caisse et cotisations Mobile Money (U4), moteur d&apos;événements (U2). Cette interface posera les bases
   252→            de chacun des 7 univers.
   253→          </p>
   254→        </div>
   255→      </FcCard>
   256→    </div>
   257→  );
   258→}

   272→/* ─────────────────────────── Notifications ─────────────────────────── */
   273→
   274→function NotifTab({ session }: { session: SessionData }) {
   275→  const [items, setItems] = useState<NotificationItem[]>([]);
   276→  const [loading, setLoading] = useState(true);
   277→  const [prefs, setPrefs] = useState<{ digestEnabled: boolean; quietStart: number; quietEnd: number; pushEnabled: boolean } | null>(null);
   278→
   279→  const load = useCallback(async () => {
   280→    setLoading(true);
   281→    const res = await api<{ notifications: NotificationItem[]; prefs: typeof prefs }>("/api/notifications");
   282→    if (res.ok) {
   283→      setItems(res.notifications);
   284→      setPrefs(res.prefs);
   285→    }
   286→    setLoading(false);
   287→  }, []);
   288→
   289→  useEffect(() => {
   290→    const t = setTimeout(load, 0);
   291→    return () => clearTimeout(t);
   292→  }, [load]);
   293→
   294→  const markRead = async (id: string) => {
   295→    await api("/api/notifications", { body: { action: "read", id } });
   296→    setItems((prev) => prev.map((n) => (n.id === id ? { ...n, read: true } : n)));
   297→  };
   298→
   299→  const markAll = async () => {
   300→    const res = await api<{ readAll: number }>("/api/notifications", { body: { action: "read-all" } });
   301→    if (res.ok) {
   302→      setItems((prev) => prev.map((n) => ({ ...n, read: true })));
   303→      toast.success(`${res.readAll} notification(s) marquée(s) comme lue(s)`);
   304→    }
   305→  };
   306→
   307→  const toggleDigest = async () => {
   308→    if (!prefs) return;
   309→    const res = await api<{ prefs: typeof prefs }>("/api/notifications", {
   310→      body: { action: "prefs", digestEnabled: !prefs.digestEnabled },
   311→    });
   312→    if (res.ok) {
   313→      setPrefs(res.prefs);
   314→      toast.success(res.prefs.digestEnabled ? "Résumé quotidien 19h activé" : "Résumé quotidien désactivé");
   315→    }
   316→  };
   317→
   318→  return (
   319→    <div className="space-y-4">
   320→      <FcCard className="p-5">
   321→        <div className="mb-3 flex items-center justify-between">
   322→          <div className="flex items-center gap-2">
   323→            <Bell className="h-5 w-5 text-[#C89F62]" />
   324→            <h3 className="fc-serif text-lg font-bold text-[#1E4D2B]">Centre d&apos;alertes</h3>
   325→          </div>
   326→          <button
   327→            onClick={markAll}
   328→            className="flex items-center gap-1 text-[12px] font-medium text-[#2A4A3A] hover:underline"
   329→          >
   330→            <CheckCheck className="h-4 w-4" /> Tout lire
   331→          </button>
   332→        </div>
   333→
   334→        {loading ? (
   335→          <p className="py-6 text-center text-sm text-[#8A9288]">Chargement…</p>
   336→        ) : items.length === 0 ? (
   337→          <p className="py-6 text-center text-sm text-[#8A9288]">
   338→            Aucune notification pour le moment. Les événements de la famille apparaîtront ici.
   339→          </p>
   340→        ) : (
   341→          <ul className="fc-scroll max-h-96 space-y-2 overflow-y-auto pr-1">
   342→            {items.map((n) => (
   343→              <li
   344→                key={n.id}
   345→                className={`rounded-2xl border p-3 transition-colors ${
   346→                  n.read ? "border-[#E5DCC0] bg-[#FFFDF6]" : "border-[#C89F62]/50 bg-[#C89F62]/10"
   347→                }`}
   348→              >
   349→                <div className="mb-1 flex items-start justify-between gap-2">
   350→                  <p className="text-[14px] font-bold text-[#1E4D2B]">{n.title}</p>
   351→                  <PriorityBadge priority={n.priority} />
   352→                </div>
   353→                <p className="text-[13px] leading-snug text-[#5C6B60]">{n.body}</p>
   354→                <div className="mt-2 flex items-center justify-between">
   355→                  <span className="text-[11px] text-[#8A9288]">
   356→                    {n.univers} · {timeAgo(n.createdAt)}
   357→                  </span>
   358→                  {!n.read && (
   359→                    <button
   360→                      onClick={() => markRead(n.id)}
   361→                      className="flex items-center gap-1 text-[11px] font-medium text-[#2A4A3A] hover:underline"
   362→                    >
   363→                      <Check className="h-3 w-3" /> Marquer lue
   364→                    </button>
   365→                  )}
   366→                </div>
   367→              </li>
   368→            ))}
   369→          </ul>
   370→        )}
   371→      </FcCard>
   372→
   373→      {prefs && (
   374→        <FcCard className="p-5">
   375→          <h3 className="fc-serif mb-3 text-lg font-bold text-[#1E4D2B]">Préférences</h3>
   376→          <label className="flex items-center justify-between gap-3 text-[14px] text-[#2B3530]">
   377→            <span>
   378→              Résumé quotidien à 19 h
   379→              <span className="block text-[12px] text-[#8A9288]">Anti-saturation : 1 récapitulatif au lieu de multiples alertes</span>
   380→            </span>
   381→            <input
   382→              type="checkbox"
   383→              checked={prefs.digestEnabled}
   384→              onChange={toggleDigest}
   385→              className="h-5 w-5 accent-[#1F5B3F]"
   386→            />
   387→          </label>
   388→          <p className="mt-3 rounded-xl bg-[#3D6B52]/5 p-3 text-[12px] leading-relaxed text-[#5C6B60]">
   389→            🔴 Les alertes <strong>critiques</strong> (décès, caisse en déficit) contournent toujours le silence
   390→            nocturne et le résumé — conformément au règlement de la famille.
   391→          </p>
   392→        </FcCard>
   393→      )}
   394→    </div>
   395→  );
   396→}

   398→/* ─────────────────────────── Sécurité ─────────────────────────── */
   399→
   400→function SecurityTab({ session, onLoggedOut }: { session: SessionData; onLoggedOut: () => void }) {
   401→  const [devices, setDevices] = useState<DeviceItem[]>([]);
   402→  const [logs, setLogs] = useState<LogItem[]>([]);
   403→  const [loading, setLoading] = useState(true);
   404→
   405→  const load = useCallback(async () => {
   406→    setLoading(true);
   407→    const [d, l] = await Promise.all([
   408→      api<{ devices: DeviceItem[] }>("/api/devices"),
   409→      api<{ logs: LogItem[] }>("/api/security-log"),
   410→    ]);
   411→    if (d.ok) setDevices(d.devices);
   412→    if (l.ok) setLogs(l.logs);
   413→    setLoading(false);
   414→  }, []);
   415→
   416→  useEffect(() => {
   417→    const t = setTimeout(load, 0);
   418→    return () => clearTimeout(t);
   419→  }, [load]);
   420→
   421→  const revoke = async (dev: DeviceItem) => {
   422→    const res = await api("/api/devices", { body: { action: "revoke", id: dev.id } });
   423→    if (res.ok) {
   424→      toast.success(`« ${dev.name} » n'a plus accès à votre compte.`);
   425→      load();
   426→    } else {
   427→      toast.error(res.error);
   428→    }
   429→  };
   430→
   431→  const logoutWeb = async () => {
   432→    const res = await api("/api/auth/logout", { method: "POST" });
   433→    if (res.ok) {
   434→      toast.success("Session fermée sur cet appareil.");
   435→      onLoggedOut();
   436→    }
   437→  };
   438→
   439→  return (
   440→    <div className="space-y-4">
   441→      <FcCard className="p-5">
   442→        <div className="mb-3 flex items-center gap-2">
   443→          <Smartphone className="h-5 w-5 text-[#C89F62]" />
   444→          <h3 className="fc-serif text-lg font-bold text-[#1E4D2B]">Mes appareils</h3>
   445→        </div>
   446→        {loading ? (
   447→          <p className="py-4 text-center text-sm text-[#8A9288]">Chargement…</p>
   448→        ) : (
   449→          <ul className="fc-scroll max-h-80 space-y-2 overflow-y-auto pr-1">
   450→            {devices.map((d) => (
   451→              <li
   452→                key={d.id}
   453→                className={`flex items-center justify-between gap-3 rounded-2xl border p-3 ${
   454→                  d.revoked ? "border-[#E5DCC0] bg-[#FFFDF6] opacity-60" : "border-[#E5DCC0] bg-[#FFFDF6]"
   455→                }`}
   456→              >
   457→                <div className="min-w-0">
   458→                  <p className="truncate text-[14px] font-semibold text-[#2B3530]">
   459→                    {d.name}

   518→        <FcButton variant="outline" onClick={logoutWeb}>

   521→      </FcCard>

   555→function AdminTab({ session }: { session: SessionData }) {
   556→  const [members, setMembers] = useState<AdminMember[]>([]);
   557→  const [branches, setBranches] = useState<{ id: string; name: string }[]>([]);
   558→  const [loading, setLoading] = useState(true);
   559→  const [fullName, setFullName] = useState("");
   560→  const [phone, setPhone] = useState("");
   561→  const [email, setEmail] = useState("");
   562→  const [branchId, setBranchId] = useState("");
   563→  const [role, setRole] = useState("ADULTE");
   564→  const [busy, setBusy] = useState(false);
   565→  const [lastCode, setLastCode] = useState<{ code: string; name: string } | null>(null);
   566→  const [copied, setCopied] = useState(false);
   567→
   568→  const load = useCallback(async () => {
   569→    setLoading(true);
   570→    const res = await api<{ members: AdminMember[]; branches: { id: string; name: string }[] }>("/api/admin/members");
   571→    if (res.ok) {
   572→      setMembers(res.members);
   573→      setBranches(res.branches);
   574→      setBranchId((prev) => prev || res.branches[0]?.id || "");
   575→    }
   576→    setLoading(false);
   577→  }, []);
   578→
   579→  useEffect(() => {
   580→    const t = setTimeout(load, 0);
   581→    return () => clearTimeout(t);
   582→  }, [load]);
   583→
   584→  const preRegister = async () => {
   585→    setBusy(true);
   586→    try {
   587→      const res = await api<{ member: { fullName: string }; accessCode: { code: string } }>("/api/admin/members", {
   588→        body: { fullName, phone, email, branchId, role },
   589→      });
   590→      if (!res.ok) throw new Error(res.error);
   591→      toast.success(`${res.member.fullName} pré-enregistré ✓ Code remis à usage unique.`);
   592→      setLastCode({ code: res.accessCode.code, name: res.member.fullName });
   593→      setFullName("");
   594→      setPhone("");
   595→      setEmail("");
   596→      load();
   597→    } catch (e) {
   598→      toast.error(e instanceof Error ? e.message : "Pré-enregistrement impossible.");
   599→    } finally {
   600→      setBusy(false);
   601→    }
   602→  };
   603→
   604→  const regenerate = async (m: AdminMember) => {
   605→    const res = await api<{ accessCode: { code: string } }>("/api/admin/members", {
   606→      method: "PUT",
   607→      body: { memberId: m.id },
   608→    });
   609→    if (res.ok) {
   610→      toast.success(`Nouveau code généré pour ${m.fullName}`);
   611→      setLastCode({ code: res.accessCode.code, name: m.fullName });
   612→      load();
   613→    } else {
   614→      toast.error(res.error);
   615→    }
   616→  };
   617→
   618→  const copyCode = async (code: string) => {
   619→    await navigator.clipboard.writeText(code).catch(() => {});
   620→    setCopied(true);
   621→    toast.success("Code copié — transmettez-le à usage unique au membre.");
   622→    setTimeout(() => setCopied(false), 2000);
   623→  };
   624→
   625→  return (
   626→    <div className="space-y-4">
   627→      <FcCard className="p-5">
   628→        <div className="mb-1 flex items-center gap-2">
   629→          <UserPlus className="h-5 w-5 text-[#C89F62]" />
   630→          <h3 className="fc-serif text-lg font-bold text-[#1E4D2B]">Pré-enregistrer un membre</h3>
   631→        </div>
   632→        <p className="mb-4 text-[13px] text-[#5C6B60]">
   633→          Remplissez ses informations : le membre n&apos;aura qu&apos;à coller son code pour un accès définitif.
   634→        </p>
   635→        <div className="space-y-2">
   636→          <input
   637→            value={fullName}
   638→            onChange={(e) => setFullName(e.target.value)}
   639→            placeholder="Nom complet *"
   640→            className="h-[48px] w-full rounded-xl border border-[#C9B98A] bg-[#FFFDF6] px-4 text-[15px] focus:border-[#2A4A3A] focus:outline-none"
   641→            aria-label="Nom complet"
   642→          />
   643→          <div className="grid grid-cols-2 gap-2">
   644→            <input
   645→              value={phone}
   646→              onChange={(e) => setPhone(e.target.value)}
   647→              placeholder="+237 6…"
   648→              inputMode="tel"
   649→              className="h-[48px] w-full rounded-xl border border-[#C9B98A] bg-[#FFFDF6] px-4 text-[15px] focus:border-[#2A4A3A] focus:outline-none"
   650→              aria-label="Téléphone"
   651→            />
   652→            <input
   653→              value={email}
   654→              onChange={(e) => setEmail(e.target.value)}
   655→              placeholder="Email (diaspora)"
   656→              inputMode="email"
   657→              className="h-[48px] w-full rounded-xl border border-[#C9B98A] bg-[#FFFDF6] px-4 text-[15px] focus:border-[#2A4A3A] focus:outline-none"
   658→              aria-label="Email"
   659→            />
   660→          </div>
   661→          <div className="grid grid-cols-2 gap-2">
   662→            <select
   663→              value={branchId}
   664→              onChange={(e) => setBranchId(e.target.value)}
   665→              className="h-[48px] w-full rounded-xl border border-[#C9B98A] bg-[#FFFDF6] px-3 text-[14px] focus:border-[#2A4A3A] focus:outline-none"
   666→              aria-label="Branche"
   667→            >
   668→              {branches.map((b) => (
   669→                <option key={b.id} value={b.id}>{b.name}</option>
