   132→   132→export function secondSunday(year: number, month0: number): number {
   133→   133→  const dow = new Date(year, month0, 1).getDay(); // 0 = dimanche
   134→   134→  return 1 + ((7 - dow) % 7) + 7;

   159→   159→/** Construit les 12 mois de l'année : 11 tours + l'AG de décembre. */
   160→   160→export function buildAnnee(year: number, roster: Member[], today = new Date()): TourMois[] {
   161→   161→  const byTour = new Map<number, Member>();
   162→   162→  roster.forEach((m) => byTour.set(m.tour, m));
   163→   163→  const moisCourant = today.getMonth();
   164→   164→  return MOIS_FR.map((mois, m0) => {
   165→   165→    const jour = secondSunday(year, m0);
   166→   166→    const date = new Date(year, m0, jour);
   167→   167→    const ag = m0 === 11;
   168→   168→    const beneficiary = ag ? null : byTour.get(m0 + 1) ?? null;
   169→   169→    const pot = ag ? 0 : potMensuel(roster.length);
   170→   170→    const collation = ag ? 0 : collationMensuelle(roster.length);
   171→   171→    const etat: EtatMois = m0 < moisCourant ? "passe" : m0 === moisCourant ? "courant" : "futur";
   172→   172→    return {
   173→   173→      month0: m0, mois, annee: year, jour, date,
   174→   174→      iso: `${year}-${String(m0 + 1).padStart(2, "0")}-${String(jour).padStart(2, "0")}`,
   175→   175→      tour: ag ? null : m0 + 1, beneficiary, ag, pot, collation, total: pot + collation, etat,
   176→   176→    };
   177→   177→  });
   178→   178→}

   180→   180→/** État de la réunion d'un mois au niveau du jour : passée / aujourd'hui / à venir. */
   181→   181→export function etatReunion(t: TourMois, today = new Date()): "passee" | "aujourdhui" | "avenir" {
   182→   182→  const t0 = new Date(today.getFullYear(), today.getMonth(), today.getDate()).getTime();
   183→   183→  const d0 = new Date(t.annee, t.month0, t.jour).getTime();
   184→   184→  if (d0 < t0) return "passee";
   185→   185→  if (d0 === t0) return "aujourdhui";
   186→   186→  return "avenir";
   187→   187→}

   265→   265→    }
   266→   266→    const paidAt = payes[payeKey(b.id, t.annee, t.month0)] ?? null;
   267→   267→    if (paidAt) return { tour: t, member: b, cotis: TONTINE.montantCotisation, colla: TONTINE.montantCollation, total: detteParPersonne(), statut: "paye" as StatutVersement, paidAt };
   268→   268→    if (t.iso <= todayIso) return { tour: t, member: b, cotis: TONTINE.montantCotisation, colla: TONTINE.montantCollation, total: detteParPersonne(), statut: "a_payer" as StatutVersement, paidAt: null };
   269→   269→    return { tour: t, member: b, cotis: TONTINE.montantCotisation, colla: TONTINE.montantCollation, total: detteParPersonne(), statut: "avenir" as StatutVersement, paidAt: null };
   270→   270→  });
   271→   271→}

   273→   273→/** Statut de cotisation d'un membre pour la réunion du mois courant. */
   274→   274→export function statutMembreDuMois(
   275→   275→  m: Member,
   276→   276→  moisCourant: TourMois | undefined,
   277→   277→  payes: Payes,
   278→   278→  today = new Date(),
   279→   279→): { label: string; cls: string } {
   280→   280→  const neutre = "bg-[#2A4A3A]/8 text-[#5C6B60] border-[#E5DCC0]";
   281→   281→  if (!moisCourant) return { label: "—", cls: neutre };
   282→   282→  if (moisCourant.ag) return { label: `AG le ${moisCourant.jour}`, cls: "bg-[#3D6B72]/10 text-[#3D6B72] border-[#3D6B72]/30" };
   283→   283→  if (moisCourant.beneficiary?.id === m.id) {
   284→   284→    return etatReunion(moisCourant, today) === "passee"
   285→   285→      ? { label: "A reçu la réunion ✓", cls: "bg-[#2F6B45]/12 text-[#2F6B45] border-[#2F6B45]/30" }

   289→   289→    return { label: `Réunion le ${moisCourant.jour}`, cls: neutre };
   290→   290→  }
   291→   291→  if (m.isVous) {
   292→   292→    const paid = !!payes[payeKey(m.id, moisCourant.annee, moisCourant.month0)];
   293→   293→    return paid
   294→   294→      ? { label: "Payé ✓", cls: "bg-[#2F6B45]/12 text-[#2F6B45] border-[#2F6B45]/30" }

   296→   296→  }
   297→   297→  // Membres fictifs — registre du trésorier : déterministe (≈ 1 sur 4 en attente).
   298→   298→  const paye = (m.tour + moisCourant.month0) % 4 !== 0;
   299→   299→  return paye
   300→   300→    ? { label: "Payé ✓", cls: "bg-[#2F6B45]/12 text-[#2F6B45] border-[#2F6B45]/30" }
   301→   301→    : { label: "En attente", cls: "bg-[#C89F62]/18 text-[#8A6420] border-[#C89F62]/45" };
   302→   302→}

   331→   331→  const n = roster.length;

   333→   333→  return annee.map((t) => {
   334→   334→    const penalites = t.etat === "futur" ? 0 : ANOMALIES.filter((a) => a.month0 === t.month0).reduce((s, a) => s + a.montant, 0);
   335→   335→    if (t.ag) {
   336→   336→      return { tour: t, encaisse: 0, pot: 0, collation: 0, caisseSociale: caisse, penalites: 0, etat: t.etat === "futur" ? ("avenir" as const) : ("cloture" as const), note: "Assemblée générale — aucun mouvement" };
   337→   337→    }
   338→   338→    if (t.etat === "futur") {
   339→   339→      return { tour: t, encaisse: t.total, pot: t.pot, collation: t.collation, caisseSociale: caisse, penalites: 0, etat: "avenir" as const, note: "Projection" };
   340→   340→    }
   341→   341→    caisse += penalites;
   342→   342→    return {
   343→   343→      tour: t, encaisse: t.total, pot: t.pot, collation: t.collation, caisseSociale: caisse, penalites,
   344→   344→      etat: t.etat === "courant" ? ("en_cours" as const) : ("cloture" as const),
   345→   345→      note: penalites > 0 ? `+ ${penalites.toLocaleString("fr-FR")} de pénalités` : "",
   346→   346→    };
   347→   347→  });
   348→   348→}

   360→   360→/** Classement des cotisants — les fictifs selon le registre, « Vous » selon votre suivi. */
   361→   361→export function classementCotisants(annee: TourMois[], roster: Member[], payes: Payes, vousId: string, today = new Date()): ClassementRow[] {
   362→   362→  const todayIso = isoDate(today);
   363→   363→  const toursEcoules = annee.filter((t) => !t.ag && t.iso <= todayIso);
   364→   364→  return roster.map((m) => {
   365→   365→    const mes = toursEcoules.filter((t) => t.beneficiary?.id !== m.id);
   366→   366→    const parts = mes.length;
   367→   367→    const anoms = ANOMALIES.filter((a) => a.memberId === m.id && mes.some((t) => t.month0 === a.month0));
   368→   368→    const versees = m.isVous
   369→   369→      ? mes.filter((t) => payes[payeKey(m.id, t.annee, t.month0)]).length
   370→   370→      : parts; // registre officiel : versé, même avec pénalité de retard
   371→   371→    const ponctualite = parts > 0 ? Math.round((versees / parts) * 100) : null;
   372→   372→    return {
   373→   373→      member: m, parts, versees,
   374→   374→      retards: anoms.length,
   375→   375→      penalites: anoms.reduce((s, a) => s + a.montant, 0),
   376→   376→      ponctualite,
   377→   377→      totalVerse: versees * detteParPersonne(),
   378→   378→    };
   379→   379→  }).sort((a, b) => {
   380→   380→    const pa = a.ponctualite ?? -1;
   381→   381→    const pb = b.ponctualite ?? -1;
   382→   382→    if (pb !== pa) return pb - pa;
   383→   383→    if (b.totalVerse !== a.totalVerse) return b.totalVerse - a.totalVerse;
   384→   384→    return a.member.tour - b.member.tour;
   385→   385→  });
   386→   386→}

   397→   397→    totalCollecte: nbTours * (pot + colla),
   398→   398→    totalPots: nbTours * pot,
   399→   399→    totalCollations: nbTours * colla,
   400→   400→    caisseSociale: roster.length * TONTINE.fraisAdhesion + penalites,
   401→   401→    penalites,
   402→   402→  };
   403→   403→}

   497→   497→/** Intérêt d'UN mois d'épargne : capital × taux (base constante — non cumulatif). */
   498→   498→export const interetEpargneMois = (capital: number, tauxPct: number) => Math.round(capital * (tauxPct / 100));
   499→   499→
   500→   500→/** Solde d'une épargne après n mois comptés : capital + n × intérêt mensuel. */
   501→   501→export const soldeEpargneAuMois = (capital: number, tauxPct: number, n: number) =>
   502→   502→  n <= 0 ? capital : capital + interetEpargneMois(capital, tauxPct) * n;
   503→   503→
   504→   504→/** Dette d'un prêt au cours du mois `cible` (hors remboursement).
   505→   505→ *  Règle du bureau : la dette gagne taux × BASE DU BLOC chaque mois, dès le
   506→   506→ *  mois du prêt, à montant constant pendant 3 mois ; à chaque échéance des
   507→   507→ *  3 mois non remboursée, la dette totale devient la nouvelle base — le
   508→   508→ *  recalcul se fait TOUS LES 3 MOIS, jamais chaque mois.
   509→   509→ *  10 000 en janvier à 3 % : janv. 10 000 · févr. 10 300 · mars 10 600
   510→   510→ *  (échéance 10 900) · avril 10 900 · mai 11 227 · juin 11 554 (échéance
   511→   511→ *  11 881)… `cible = 12` → solde final de fin décembre (14 113). */
   512→   512→export function dettePretAuMois(capital: number, tauxPct: number, depart: number, cible: number): number {
   513→   513→  const n = Math.min(Math.max(cible - depart, 0), 12 - depart); // mois d'intérêts écoulés (le mois du prêt compte, plafonné à fin décembre)
   514→   514→  if (n <= 0) return capital;
   515→   515→  const r = tauxPct / 100;
   516→   516→  let base = capital; // base du bloc courant — recalculée seulement tous les 3 mois
   517→   517→  let solde = capital;
   518→   518→  for (let i = 1; i <= n; i++) {
   519→   519→    solde += Math.round(base * r); // intérêt du mois — constant pendant les 3 mois du bloc
   520→   520→    if (i % 3 === 0) base = solde; // échéance des 3 mois : la dette devient la nouvelle base
   521→   521→  }
   522→   522→  return solde;
   523→   523→}

   525→   525→/** Nombre de réunions déjà tenues APRÈS le mois de départ (l'épargne commence
   526→   526→ *  à courir le mois suivant) — plafonné à décembre. */
   527→   527→export function moisEcoules(depart: number, annee: TourMois[], today = new Date()): number {
   528→   528→  const todayIso = isoDate(today);

   530→   530→  for (let m = depart + 1; m <= 11; m++) {
   531→   531→    if (annee[m] && annee[m].iso <= todayIso) n++;
   532→   532→  }
   533→   533→  return n;
   534→   534→}

   538→   538→ *  calendrier absent. Utilisé par les prêts (le mois du prêt compte). */
   539→   539→export function moisCourantAnnee(annee: TourMois[], today = new Date()): number {

   544→   544→  }

   546→   546→}
