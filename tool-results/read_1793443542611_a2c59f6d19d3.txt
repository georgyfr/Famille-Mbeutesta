   275→function NotifTab({ session }: { session: SessionData }) {
   276→  const [items, setItems] = useState<NotificationItem[]>([]);
   277→  const [loading, setLoading] = useState(true);
   278→  const [prefs, setPrefs] = useState<{ digestEnabled: boolean; quietStart: number; quietEnd: number; pushEnabled: boolean } | null>(null);
   279→
   280→  const load = useCallback(async () => {
   281→    setLoading(true);
   282→    const res = await api<{ notifications: NotificationItem[]; prefs: typeof prefs }>("/api/notifications");
   283→    if (res.ok) {
   284→      setItems(res.notifications);
   285→      setPrefs(res.prefs);
   286→    }
   287→    setLoading(false);
   288→  }, []);
   289→
   290→  useEffect(() => {
   291→    const t = setTimeout(load, 0);
   292→    return () => clearTimeout(t);
   293→  }, [load]);
   294→
   295→  const markRead = async (id: string) => {
   296→    await api("/api/notifications", { body: { action: "read", id } });
   297→    setItems((prev) => prev.map((n) => (n.id === id ? { ...n, read: true } : n)));
   298→  };
   299→
   300→  const markAll = async () => {
   301→    const res = await api<{ readAll: number }>("/api/notifications", { body: { action: "read-all" } });
   302→    if (res.ok) {
   303→      setItems((prev) => prev.map((n) => ({ ...n, read: true })));
   304→      toast.success(`${res.readAll} notification(s) marquée(s) comme lue(s)`);
   305→    }
   306→  };
   307→
   308→  const toggleDigest = async () => {
   309→    if (!prefs) return;
   310→    const res = await api<{ prefs: typeof prefs }>("/api/notifications", {
   311→      body: { action: "prefs", digestEnabled: !prefs.digestEnabled },
   312→    });
   313→    if (res.ok) {
   314→      setPrefs(res.prefs);
   315→      toast.success(res.prefs.digestEnabled ? "Résumé quotidien 19h activé" : "Résumé quotidien désactivé");
   316→    }
   317→  };
   318→
   319→  return (
   320→    <div className="space-y-4">
   321→      <FcCard className="p-5">
   322→        <div className="mb-3 flex items-center justify-between">
   323→          <div className="flex items-center gap-2">
   324→            <Bell className="h-5 w-5 text-[#C89F62]" />
   325→            <h3 className="fc-serif text-lg font-bold text-[#1E4D2B]">Centre d&apos;alertes</h3>
   326→          </div>
   327→          <button
   328→            onClick={markAll}
   329→            className="flex items-center gap-1 text-[12px] font-medium text-[#2A4A3A] hover:underline"
   330→          >
   331→            <CheckCheck className="h-4 w-4" /> Tout lire
   332→          </button>
   333→        </div>
   334→
   335→        {loading ? (
   336→          <p className="py-6 text-center text-sm text-[#8A9288]">Chargement…</p>
   337→        ) : items.length === 0 ? (
   338→          <p className="py-6 text-center text-sm text-[#8A9288]">
   339→            Aucune notification pour le moment. Les événements de la famille apparaîtront ici.
   340→          </p>
   341→        ) : (
   342→          <ul className="fc-scroll max-h-96 space-y-2 overflow-y-auto pr-1">
   343→            {items.map((n) => (
   344→              <li
   345→                key={n.id}
   346→                className={`rounded-2xl border p-3 transition-colors ${
   347→                  n.read ? "border-[#E5DCC0] bg-[#FFFDF6]" : "border-[#C89F62]/50 bg-[#C89F62]/10"
   348→                }`}
   349→              >
   350→                <div className="mb-1 flex items-start justify-between gap-2">
   351→                  <p className="text-[14px] font-bold text-[#1E4D2B]">{n.title}</p>
   352→                  <PriorityBadge priority={n.priority} />
   353→                </div>
   354→                <p className="text-[13px] leading-snug text-[#5C6B60]">{n.body}</p>
   355→                <div className="mt-2 flex items-center justify-between">
   356→                  <span className="text-[11px] text-[#8A9288]">
   357→                    {n.univers} · {timeAgo(n.createdAt)}
   358→                  </span>
   359→                  {!n.read && (
   360→                    <button
   361→                      onClick={() => markRead(n.id)}
   362→                      className="flex items-center gap-1 text-[11px] font-medium text-[#2A4A3A] hover:underline"
   363→                    >
   364→                      <Check className="h-3 w-3" /> Marquer lue
   365→                    </button>
   366→                  )}
   367→                </div>
   368→              </li>
   369→            ))}
   370→          </ul>
   371→        )}
   372→      </FcCard>
   373→
   374→      {prefs && (
   375→        <FcCard className="p-5">
   376→          <h3 className="fc-serif mb-3 text-lg font-bold text-[#1E4D2B]">Préférences</h3>
   377→          <label className="flex items-center justify-between gap-3 text-[14px] text-[#2B3530]">
   378→            <span>
   379→              Résumé quotidien à 19 h
   380→              <span className="block text-[12px] text-[#8A9288]">Anti-saturation : 1 récapitulatif au lieu de multiples alertes</span>
   381→            </span>
   382→            <input
   383→              type="checkbox"
   384→              checked={prefs.digestEnabled}
   385→              onChange={toggleDigest}
   386→              className="h-5 w-5 accent-[#1F5B3F]"
   387→            />
   388→          </label>
   389→          <p className="mt-3 rounded-xl bg-[#3D6B52]/5 p-3 text-[12px] leading-relaxed text-[#5C6B60]">
   390→            🔴 Les alertes <strong>critiques</strong> (décès, caisse en déficit) contournent toujours le silence
   391→            nocturne et le résumé — conformément au règlement de la famille.
   392→          </p>
   393→        </FcCard>
   394→      )}
   395→    </div>
   396→  );
   397→}

   409→      (navigator.platform === "MacIntel" && navigator.maxTouchPoints > 1))
   410→  );
   411→
   412→  useEffect(() => {
   413→    const t = setTimeout(async () => setState(await getPushState()), 0);
   414→    return () => clearTimeout(t);
   415→  }, []);
   416→
   417→  const enable = async () => {
   418→    setBusy(true);
   419→    const ok = await enablePush();
   420→    if (ok) setState(await getPushState());
   421→    setBusy(false);
   422→  };
   423→
   424→  const verify = async () => {
   425→    setBusy(true);
   426→    const res = await testPush();
   427→    if (res.ok) {
   428→      toast.success("Notification envoyée — regardez votre écran, même vérrouillé 🔔");
   429→    } else {
   430→      toast.error(res.error || "Envoi impossible.");
   431→    }
   432→    setBusy(false);
   433→  };
   434→
   435→  return (
   436→    <FcCard className="p-5">
   437→      <div className="mb-2 flex items-center gap-2">
   438→        <Bell className="h-5 w-5 text-[#C89F62]" />
   439→        <h3 className="fc-serif text-lg font-bold text-[#1E4D2B]">Notifications du téléphone</h3>
   440→      </div>
   441→      {state === "unsupported" && (
   442→        <p className="text-[13px] leading-relaxed text-[#8A9288]">
   443→          {isIOS
   444→            ? "Sur iPhone : ajoutez d'abord l'app à l'écran d'accueil (Partager → Sur l'écran d'accueil), puis ouvrez-la depuis l'icône."
   445→            : "Ce navigateur ne prend pas en charge les notifications push."}
   446→        </p>
   447→      )}
   448→      {state === "denied" && (
   449→        <p className="flex items-center gap-2 text-[13px] text-[#B91C1C]">
   450→          <BellOff className="h-4 w-4" /> Notifications bloquées — réautorisez-les dans les réglages du site
   451→          (icône cadenas dans la barre d'adresse).
   452→        </p>
   453→      )}
   454→      {(state === "default" || state === "granted") && (
   455→        <>
   456→          <p className="mb-3 text-[13px] leading-relaxed text-[#5C6B60]">
   457→            Recevez les événements de la famille même application fermée — avec le son et la vibration,
   458→            comme WhatsApp. Gratuit et illimité.
   459→          </p>
   460→          <FcButton onClick={enable} disabled={busy}>
   461→            {busy ? "Activation…" : "Activer les notifications"}
