   167→/* ───────────────────────── Écran d'accueil ───────────────────────── */
   168→
   169→function WelcomeScreen({
   170→  remembered,
   171→  busy,
   172→  setBusy,
   173→  onReady,
   174→  goto,
   175→}: {
   176→  remembered: { uid: string; name: string; hasPasskey: boolean } | null;
   177→  busy: boolean;
   178→  setBusy: (b: boolean) => void;
   179→  onReady: (name: string) => void;
   180→  goto: (s: Step) => void;
   181→}) {
   182→  const handleRealPasskey = useCallback(async () => {
   183→    if (!remembered) return;
   184→    setBusy(true);
   185→    try {
   186→      const { options } = await api<{ options: unknown }>("/api/auth/passkey", {
   187→        body: { action: "auth-options" },
   188→      });
   189→      const response = await startAuthentication({ optionsJSON: options as never });
   190→      const result = await api<{ user: { fullName: string } }>("/api/auth/passkey", {
   191→        body: { action: "auth-verify", response },
   192→      });
   193→      if (!result.ok) throw new Error(result.error);
   194→      localStorage.setItem("fc_uid", remembered.uid);
   195→      localStorage.setItem("fc_name", result.user.fullName);
   196→      toast.success(`Content de vous revoir, ${result.user.fullName.split(" ")[0]} !`);
   197→      onReady(result.user.fullName);
   198→    } catch (e) {
   199→      toast.error(e instanceof Error ? e.message : "Connexion biométrique impossible sur cet appareil.");
   200→    } finally {
   201→      setBusy(false);
   202→    }
   203→  }, [remembered, onReady, setBusy]);
   204→
   205→  const handleSimulatedPasskey = useCallback(async () => {
   206→    if (!remembered) return;
   207→    setBusy(true);
   208→    try {
   209→      const result = await api<{ user: { fullName: string } }>("/api/auth/passkey", {
   210→        body: { action: "simulate", uid: remembered.uid },
   211→      });
   212→      if (!result.ok) throw new Error(result.error);
   213→      localStorage.setItem("fc_uid", remembered.uid);
   214→      localStorage.setItem("fc_name", result.user.fullName);
   215→      toast.success(`Content de vous revoir, ${result.user.fullName.split(" ")[0]} ! (biométrie simulée)`);
   216→      onReady(result.user.fullName);
   217→    } catch (e) {
   218→      toast.error(e instanceof Error ? e.message : "Échec de la connexion simulée.");
   219→    } finally {
   220→      setBusy(false);
   221→    }
   222→  }, [remembered, onReady, setBusy]);
   223→
   224→  return (
   225→    <>
   226→      <FcCard className="p-6 sm:p-8">
   227→      <h2 className="fc-serif mb-1 text-center text-3xl font-bold text-[#1E4D2B]">Bon retour 👋</h2>
   228→      <p className="mb-6 text-center text-[15px] text-[#5C6B60]">
   229→        Connectez-vous pour accéder à votre espace familial.
   230→      </p>
   231→
   232→      {remembered && (
   233→        <div className="mb-5 rounded-2xl border border-[#C89F62]/40 bg-[#C89F62]/10 p-4 text-center">
   234→          <p className="text-sm text-[#7A5B2E]">
   235→            Bon retour parmi nous, <span className="font-bold">{remembered.name}</span>
   236→          </p>
   237→        </div>
   238→      )}
   239→
   240→      {remembered && remembered.hasPasskey && (
   241→        <div className="mb-3 space-y-2">
   242→          <FcButton onClick={handleRealPasskey} disabled={busy}>
   243→            {busy ? <FcSpinner /> : <Fingerprint className="h-5 w-5" />}
   244→            Se connecter avec mon empreinte
   245→          </FcButton>
   246→          <FcButton variant="ghost" className="text-sm" onClick={handleSimulatedPasskey} disabled={busy}>
   247→            <ShieldCheck className="h-4 w-4" /> Biométrie simulée (démo)
   248→          </FcButton>
   249→          <FcDivider label="ou" />
   250→        </div>
   251→      )}
   252→
   253→      {remembered && (
   254→        <div className="mb-3 space-y-2">
   255→          <FcButton
   256→            variant="outline"
   257→            onClick={() => goto({ k: "pin-login", uid: remembered.uid, name: remembered.name, hasPasskey: remembered.hasPasskey })}
   258→          >
   259→            <Lock className="h-5 w-5" /> Mon code PIN
   260→          </FcButton>
   261→          <FcDivider label="nouvel appareil" />
   262→        </div>
   263→      )}
   264→
   265→      <div className="space-y-2">
   266→        <FcButton onClick={() => goto({ k: "code-entry" })} disabled={busy}>
   267→          <KeyRound className="h-5 w-5" /> J&apos;ai un code d&apos;accès
   268→        </FcButton>
   269→        <FcButton variant="ghost" className="text-sm" onClick={() => goto({ k: "recovery-entry" })} disabled={busy}>
   270→          <LifeBuoy className="h-4 w-4" /> Récupérer mon accès (code personnel)
   271→        </FcButton>
   272→      </div>
   273→
   274→      <p className="mt-5 rounded-xl bg-[#3D6B52]/5 p-3 text-center text-[13px] leading-relaxed text-[#5C6B60]">
   275→        Pas encore de code ? Le <strong>secrétaire de famille</strong> pré-enregistre chaque membre et remet
   276→        un code d&apos;accès. Rendez-vous à la prochaine réunion, ou contactez-le.
   277→      </p>
   278→      </FcCard>
   279→
   280→      {/* Encart sécurité — fidèle à la maquette validée */}
   281→      <div className="flex gap-3 rounded-2xl border border-[#C89F62]/30 bg-[#C89F62]/10 p-4 text-[13px] leading-relaxed text-[#5C6B60]">

   292→/* ───────────────────────── Saisie du code d'accès ───────────────────────── */
   293→
   294→function CodeEntryScreen({
   295→  busy,
   296→  setBusy,
   297→  goto,
   298→}: {
   299→  busy: boolean;
   300→  setBusy: (b: boolean) => void;
   301→  goto: (s: Step) => void;
   302→}) {
   303→  const [code, setCode] = useState("");
   304→  const handle = async () => {
   305→    if (code.trim().length < 4) {
   306→      toast.error("Entrez le code d'accès remis par le secrétaire (ex : FC-XXXX-XXXX).");
   307→      return;
   308→    }
   309→    setBusy(true);
   310→    try {
   311→      const res = await api<{ claimToken: string; member: MemberInfo }>("/api/auth/verify-code", {
   312→        body: { code, deviceName: deviceName() },
   313→      });
   314→      if (!res.ok) throw new Error(res.error);
   315→      toast.success("Code validé ✓");
   316→      goto({ k: "claim-confirm", claimToken: res.claimToken, member: res.member });
   317→    } catch (e) {
   318→      toast.error(e instanceof Error ? e.message : "Code invalide.");
   319→    } finally {
   320→      setBusy(false);
   321→    }
   322→  };
   323→  return (
   324→    <FcCard className="p-6 sm:p-8">
   325→      <BackButton onClick={() => goto({ k: "welcome" })} />
   326→      <h2 className="fc-serif mb-1 text-center text-2xl font-bold text-[#1E4D2B]">Code d&apos;accès</h2>
   327→      <p className="mb-6 text-center text-[14px] text-[#5C6B60]">
   328→        Saisissez le code remis par le secrétaire de famille.
   329→      </p>
   330→      <FcInput
   331→        icon={<KeyRound className="h-5 w-5" />}
   332→        placeholder="FC-XXXX-XXXX"
   333→        value={code}
   334→        onChange={(e) => setCode(e.target.value.toUpperCase())}
   335→        onKeyDown={(e) => e.key === "Enter" && handle()}
   336→        autoCapitalize="characters"
   337→        autoComplete="one-time-code"
   338→        className="mb-4"
   339→        aria-label="Code d'accès"
   340→      />
   341→      <FcButton onClick={handle} disabled={busy}>
   342→        {busy ? <FcSpinner /> : "Vérifier mon code"}
   343→      </FcButton>
   344→    </FcCard>
   345→  );
   346→}

   404→/* ───────────────────────── Vérification OTP (mode démo) ───────────────────────── */
   405→
   406→function OtpScreen({
   407→  state,
   408→  busy,
   409→  setBusy,
   410→  goto,
   411→  onReady,
   412→}: {
   413→  state: Extract<Step, { k: "otp" }>;
   414→  busy: boolean;
   415→  setBusy: (b: boolean) => void;
   416→  goto: (s: Step) => void;
   417→  onReady: (name: string) => void;
   418→}) {
   419→  const [code, setCode] = useState("");
   420→  const [demoCode, setDemoCode] = useState<string | undefined>(state.demoCode);
   421→  const requested = useRef(false);
   422→
   423→  const requestCode = useCallback(async () => {
   424→    setBusy(true);
   425→    try {
   426→      const res = await api<{ demoMode: boolean; demoCode?: string }>("/api/auth/otp", {
   427→        body: { action: "request", claimToken: state.claimToken },
   428→      });
   429→      if (!res.ok) throw new Error(res.error);
   430→      if (res.demoCode) {
   431→        setDemoCode(res.demoCode);
   432→        toast.info(`Mode démo — votre code de vérification : ${res.demoCode}`);
   433→      } else {
   434→        toast.success("Code de vérification envoyé par SMS.");
   435→      }
   436→    } catch (e) {
   437→      toast.error(e instanceof Error ? e.message : "Envoi du code impossible.");
   438→    } finally {
   439→      setBusy(false);
   440→    }
   441→  }, [state.claimToken, setBusy]);
   442→
   443→  useEffect(() => {
   444→    if (!requested.current) {
   445→      requested.current = true;
   446→      requestCode();
   447→    }
   448→  }, [requestCode]);
   449→
   450→  const verify = async () => {
   451→    setBusy(true);
   452→    try {
   453→      const res = await api<{ activateToken: string }>("/api/auth/otp", {
   454→        body: { action: "verify", claimToken: state.claimToken, code },
   455→      });
   456→      if (!res.ok) throw new Error(res.error);
   457→      const activation = await api<{ recoveryCode: string; user: { fullName: string }; reenroll: boolean }>(
   458→        "/api/auth/activate",
   459→        { body: { activateToken: res.activateToken } }
   460→      );
   461→      if (!activation.ok) throw new Error(activation.error);
   462→      localStorage.setItem("fc_uid", activation.user.id);
   463→      localStorage.setItem("fc_name", activation.user.fullName);
   464→      localStorage.removeItem("fc_has_passkey");
   465→      toast.success(activation.reenroll ? "Accès réactivé ✓" : "Compte activé ✓ Bienvenue !");
   466→      goto({ k: "setup", recoveryCode: activation.recoveryCode, userName: activation.user.fullName, reenroll: activation.reenroll });
   467→    } catch (e) {
   468→      toast.error(e instanceof Error ? e.message : "Code incorrect.");
   469→    } finally {
   470→      setBusy(false);
   471→    }
   472→  };
   473→
   474→  return (
   475→    <FcCard className="p-6 sm:p-8">
   476→      <BackButton onClick={() => goto({ k: "code-entry" })} />
   477→      <h2 className="fc-serif mb-1 text-center text-2xl font-bold text-[#1E4D2B]">Vérification</h2>
   478→      <p className="mb-5 text-center text-[14px] text-[#5C6B60]">
   479→        Entrez le code à 6 chiffres envoyé pour {state.member.fullName}.
   480→      </p>
   481→

   643→
   644→function SetupScreen({
   645→  state,
   646→  busy,
   647→  setBusy,
   648→  onReady,
   649→}: {
   650→  state: Extract<Step, { k: "setup" }>;
   651→  busy: boolean;
   652→  setBusy: (b: boolean) => void;
   653→  onReady: (name: string) => void;
   654→}) {
   655→  const [ack, setAck] = useState(false);
   656→  const [copied, setCopied] = useState(false);
   657→  const [pin, setPin] = useState("");
   658→  const [pinDone, setPinDone] = useState(false);
   659→  const [bioDone, setBioDone] = useState(false);
   660→  const [bioSimulated, setBioSimulated] = useState(false);
   661→
   662→  const copyRecovery = async () => {
   663→    await navigator.clipboard.writeText(state.recoveryCode).catch(() => {});
   664→    setCopied(true);
   665→    toast.success("Code de récupération copié — collez-le dans un endroit sûr.");
   666→  };
   667→
   668→  const savePin = async () => {
   669→    setBusy(true);
   670→    try {
   671→      const res = await api("/api/auth/pin", { body: { action: "setup", pin } });
   672→      if (!res.ok) throw new Error(res.error);
   673→      setPinDone(true);
   674→      toast.success("Code PIN enregistré ✓");
   675→    } catch (e) {
   676→      toast.error(e instanceof Error ? e.message : "Échec de l'enregistrement du PIN.");
   677→    } finally {
   678→      setBusy(false);
   679→    }
   680→  };
   681→
   682→  const enrollBiometry = async () => {
   683→    setBusy(true);
   684→    try {
   685→      const { options } = await api<{ options: unknown }>("/api/auth/passkey", {
   686→        body: { action: "register-options" },
   687→      });
   688→      const response = await startRegistration({ optionsJSON: options as never });
   689→      const result = await api("/api/auth/passkey", {
   690→        body: { action: "register-verify", response, deviceLabel: deviceName() },
   691→      });
   692→      if (!result.ok) throw new Error(result.error);
   693→      setBioDone(true);
   694→      localStorage.setItem("fc_has_passkey", "1");
   695→      toast.success("Biométrie activée ✓ Vous vous connecterez d'un simple geste.");
   696→    } catch (e) {
   697→      toast.error(
   698→        e instanceof Error
   699→          ? `Biométrie indisponible ici (${e.message.slice(0, 60)}). Utilisez le mode démo ci-dessous.`
   700→          : "Biométrie indisponible."
   701→      );
   702→    } finally {
   703→      setBusy(false);
   704→    }
   705→  };
   706→
   707→  const simulateBiometry = async () => {
   708→    setBusy(true);
   709→    try {
   710→      const result = await api("/api/auth/passkey", { body: { action: "simulate-register" } });
   711→      if (!result.ok) throw new Error(result.error);
   712→      setBioDone(true);
   713→      setBioSimulated(true);
   714→      localStorage.setItem("fc_has_passkey", "0");
   715→      toast.success("Appareil biométrique simulé enregistré (mode démo) ✓");
   716→    } catch (e) {
   717→      toast.error(e instanceof Error ? e.message : "Simulation impossible.");
   718→    } finally {
   719→      setBusy(false);
   720→    }
   721→  };
   722→
   723→  const enter = () => {
   724→    if (!ack) {
   725→      toast.error("Confirmez d'abord avoir noté votre code de récupération.");
   726→      return;
   727→    }
   728→    onReady(state.userName);
   729→  };
   730→
   731→  return (
   732→    <div className="space-y-4">
   733→      {/* 1. Code de récupération */}
   734→      <FcCard className="p-6">
   735→        <div className="mb-2 flex items-center gap-2">
   736→          <LifeBuoy className="h-5 w-5 text-[#D97706]" />
   737→          <h2 className="fc-serif text-xl font-bold text-[#1E4D2B]">Votre code de récupération</h2>
   738→        </div>
   739→        <p className="mb-3 text-[13px] leading-relaxed text-[#5C6B60]">
   740→          {state.reenroll
   741→            ? "Un NOUVEAU code vous a été attribué (l'ancien est invalidé)."
   742→            : "C'est la seule fois où il sera affiché."}{" "}
   743→          Notez-le sur papier ou envoyez-le-vous par WhatsApp. Il permet de récupérer votre accès si vous perdez
   744→          votre téléphone.
   745→        </p>
   746→        <div className="mb-3 flex items-center justify-center gap-3 rounded-2xl border-2 border-dashed border-[#D97706] bg-[#D97706]/5 p-4">
   747→          <span className="fc-serif text-2xl font-bold tracking-wider text-[#7A5B2E]">{state.recoveryCode}</span>
   748→          <button
   749→            onClick={copyRecovery}
   750→            className="rounded-lg p-2 text-[#C89F62] hover:bg-[#C89F62]/10"
   751→            aria-label="Copier le code de récupération"
   752→          >
   753→            {copied ? <Check className="h-5 w-5" /> : <Copy className="h-5 w-5" />}
   754→          </button>
   755→        </div>
   756→        <label className="flex cursor-pointer items-start gap-2 text-[14px] text-[#2B3530]">
   757→          <input
   758→            type="checkbox"
   759→            checked={ack}
   760→            onChange={(e) => setAck(e.target.checked)}
   761→            className="mt-0.5 h-4 w-4 accent-[#1F5B3F]"
   762→          />
   763→          <span>J&apos;ai noté mon code de récupération en lieu sûr.</span>
   764→        </label>
   765→      </FcCard>
   766→
   767→      {/* 2. PIN de secours */}
   768→      <FcCard className="p-6">
   769→        <div className="mb-2 flex items-center gap-2">
   770→          <Lock className="h-5 w-5 text-[#2A4A3A]" />
   771→          <h2 className="fc-serif text-xl font-bold text-[#1E4D2B]">Code PIN de secours</h2>
   772→        </div>
   773→        {pinDone ? (
   774→          <p className="text-sm font-medium text-[#16A34A]">✓ PIN défini — il déverrouille cet appareil.</p>
   775→        ) : (
   776→          <>
   777→            <p className="mb-3 text-[13px] text-[#5C6B60]">
   778→              4 à 6 chiffres, mémorisable. Sert quand la biométrie est indisponible. (Recommandé, mais optionnel.)
   779→            </p>
   780→            <div className="flex gap-2">
   781→              <FcInput
   782→                placeholder="● ● ● ●"
   783→                value={pin}
   784→                onChange={(e) => setPin(e.target.value.replace(/\D/g, "").slice(0, 6))}
   785→                inputMode="numeric"
   786→                type="password"
   787→                maxLength={6}
   788→                className="flex-1"
   789→                aria-label="Nouveau code PIN"
   790→              />
   791→              <FcButton onClick={savePin} disabled={busy || pin.length < 4} className="w-auto px-6">
   792→                Définir
   793→              </FcButton>
   794→            </div>
   795→          </>
   796→        )}
   797→      </FcCard>
   798→
   799→      {/* 3. Biométrie */}
   800→      <FcCard className="p-6">
   801→        <div className="mb-2 flex items-center gap-2">
   802→          <Fingerprint className="h-5 w-5 text-[#2A4A3A]" />
   803→          <h2 className="fc-serif text-xl font-bold text-[#1E4D2B]">Connexion par empreinte</h2>
   804→        </div>
   805→        {bioDone ? (
   806→          <p className="text-sm font-medium text-[#16A34A]">
   807→            ✓ {bioSimulated ? "Appareil simulé enregistré (démo)" : "Biométrie active sur cet appareil"}
   808→          </p>
   809→        ) : (
   810→          <div className="space-y-2">
   811→            <p className="mb-2 text-[13px] text-[#5C6B60]">
   812→              Un geste = connecté, pour toujours. Zéro mot de passe à retenir.
   813→            </p>
   814→            <FcButton onClick={enrollBiometry} disabled={busy}>
   815→              {busy ? <FcSpinner /> : <Fingerprint className="h-5 w-5" />} Activer mon empreinte
   816→            </FcButton>
   817→            <FcButton variant="outline" onClick={simulateBiometry} disabled={busy}>
   818→              <ShieldCheck className="h-4 w-4" /> Simuler sur cet appareil (démo)
   819→            </FcButton>
   820→          </div>
   821→        )}
   822→      </FcCard>
   823→
   824→      <FcButton onClick={enter} className="h-[54px] text-[17px]">
   825→        Entrer dans Famille.Mbeutetsa
   826→      </FcButton>
   827→    </div>
   828→  );
   829→}
