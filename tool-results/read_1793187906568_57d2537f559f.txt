   266→   266→function NotifTab({ session }: { session: SessionData }) {
   267→   267→  const [items, setItems] = useState<NotificationItem[]>([]);
   268→   268→  const [loading, setLoading] = useState(true);
   269→   269→  const [prefs, setPrefs] = useState<{ digestEnabled: boolean; quietStart: number; quietEnd: number; pushEnabled: boolean } | null>(null);
   270→   270→
   271→   271→  const load = useCallback(async () => {
   272→   272→    setLoading(true);
   273→   273→    const res = await api<{ notifications: NotificationItem[]; prefs: typeof prefs }>("/api/notifications");
   274→   274→    if (res.ok) {
   275→   275→      setItems(res.notifications);
   276→   276→      setPrefs(res.prefs);
   277→   277→    }
   278→   278→    setLoading(false);
   279→   279→  }, []);
   280→   280→
   281→   281→  useEffect(() => {
   282→   282→    const t = setTimeout(load, 0);
   283→   283→    return () => clearTimeout(t);
   284→   284→  }, [load]);
   285→   285→
   286→   286→  const markRead = async (id: string) => {
   287→   287→    await api("/api/notifications", { body: { action: "read", id } });
   288→   288→    setItems((prev) => prev.map((n) => (n.id === id ? { ...n, read: true } : n)));
   289→   289→  };
   290→   290→
   291→   291→  const markAll = async () => {
   292→   292→    const res = await api<{ readAll: number }>("/api/notifications", { body: { action: "read-all" } });
   293→   293→    if (res.ok) {
   294→   294→      setItems((prev) => prev.map((n) => ({ ...n, read: true })));
   295→   295→      toast.success(`${res.readAll} notification(s) marquée(s) comme lue(s)`);
   296→   296→    }
   297→   297→  };
   298→   298→
   299→   299→  const toggleDigest = async () => {
   300→   300→    if (!prefs) return;
   301→   301→    const res = await api<{ prefs: typeof prefs }>("/api/notifications", {
   302→   302→      body: { action: "prefs", digestEnabled: !prefs.digestEnabled },
   303→   303→    });
   304→   304→    if (res.ok) {
   305→   305→      setPrefs(res.prefs);
   306→   306→      toast.success(res.prefs.digestEnabled ? "Résumé quotidien 19h activé" : "Résumé quotidien désactivé");
   307→   307→    }
   308→   308→  };
   309→   309→
   310→   310→  return (
   311→   311→    <div className="space-y-4">
   312→   312→      <FcCard className="p-5">
   313→   313→        <div className="mb-3 flex items-center justify-between">
   314→   314→          <div className="flex items-center gap-2">
   315→   315→            <Bell className="h-5 w-5 text-[#C89F62]" />
   316→   316→            <h3 className="fc-serif text-lg font-bold text-[#1E4D2B]">Centre d&apos;alertes</h3>
   317→   317→          </div>
   318→   318→          <button
   319→   319→            onClick={markAll}
   320→   320→            className="flex items-center gap-1 text-[12px] font-medium text-[#2A4A3A] hover:underline"
   321→   321→          >
   322→   322→            <CheckCheck className="h-4 w-4" /> Tout lire
   323→   323→          </button>
   324→   324→        </div>
   325→   325→
   326→   326→        {loading ? (
   327→   327→          <p className="py-6 text-center text-sm text-[#8A9288]">Chargement…</p>
   328→   328→        ) : items.length === 0 ? (
   329→   329→          <p className="py-6 text-center text-sm text-[#8A9288]">
   330→   330→            Aucune notification pour le moment. Les événements de la famille apparaîtront ici.
   331→   331→          </p>
   332→   332→        ) : (
   333→   333→          <ul className="fc-scroll max-h-96 space-y-2 overflow-y-auto pr-1">
   334→   334→            {items.map((n) => (
   335→   335→              <li
   336→   336→                key={n.id}
   337→   337→                className={`rounded-2xl border p-3 transition-colors ${
   338→   338→                  n.read ? "border-[#E5DCC0] bg-[#FFFDF6]" : "border-[#C89F62]/50 bg-[#C89F62]/10"
   339→   339→                }`}
   340→   340→              >
   341→   341→                <div className="mb-1 flex items-start justify-between gap-2">
   342→   342→                  <p className="text-[14px] font-bold text-[#1E4D2B]">{n.title}</p>
   343→   343→                  <PriorityBadge priority={n.priority} />
   344→   344→                </div>
   345→   345→                <p className="text-[13px] leading-snug text-[#5C6B60]">{n.body}</p>
   346→   346→                <div className="mt-2 flex items-center justify-between">
   347→   347→                  <span className="text-[11px] text-[#8A9288]">
   348→   348→                    {n.univers} · {timeAgo(n.createdAt)}
   349→   349→                  </span>
   350→   350→                  {!n.read && (
   351→   351→                    <button
   352→   352→                      onClick={() => markRead(n.id)}
   353→   353→                      className="flex items-center gap-1 text-[11px] font-medium text-[#2A4A3A] hover:underline"
   354→   354→                    >
   355→   355→                      <Check className="h-3 w-3" /> Marquer lue

   357→   357→                  )}
   358→   358→                </div>
   359→   359→              </li>
   360→   360→            ))}
   361→   361→          </ul>
   362→   362→        )}
   363→   363→      </FcCard>
   364→   364→
   365→   365→      {prefs && (
   366→   366→        <FcCard className="p-5">
   367→   367→          <h3 className="fc-serif mb-3 text-lg font-bold text-[#1E4D2B]">Préférences</h3>

   369→   369→            <span>
   370→   370→              Résumé quotidien à 19 h
   371→   371→              <span className="block text-[12px] text-[#8A9288]">Anti-saturation : 1 récapitulatif au lieu de multiples alertes</span>
   372→   372→            </span>
   373→   373→            <input
   374→   374→              type="checkbox"
