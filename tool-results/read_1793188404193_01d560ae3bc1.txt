   266→function NotifTab({ session }: { session: SessionData }) {
   267→  const [items, setItems] = useState<NotificationItem[]>([]);
   268→  const [loading, setLoading] = useState(true);
   269→  const [prefs, setPrefs] = useState<{ digestEnabled: boolean; quietStart: number; quietEnd: number; pushEnabled: boolean } | null>(null);
   270→
   271→  const load = useCallback(async () => {
   272→    setLoading(true);
   273→    const res = await api<{ notifications: NotificationItem[]; prefs: typeof prefs }>("/api/notifications");
   274→    if (res.ok) {
   275→      setItems(res.notifications);
   276→      setPrefs(res.prefs);
   277→    }
   278→    setLoading(false);
   279→  }, []);
   280→
   281→  useEffect(() => {
   282→    const t = setTimeout(load, 0);
   283→    return () => clearTimeout(t);
   284→  }, [load]);
   285→
   286→  const markRead = async (id: string) => {
   287→    await api("/api/notifications", { body: { action: "read", id } });
   288→    setItems((prev) => prev.map((n) => (n.id === id ? { ...n, read: true } : n)));
   289→  };
   290→
   291→  const markAll = async () => {
   292→    const res = await api<{ readAll: number }>("/api/notifications", { body: { action: "read-all" } });
   293→    if (res.ok) {
   294→      setItems((prev) => prev.map((n) => ({ ...n, read: true })));
   295→      toast.success(`${res.readAll} notification(s) marquée(s) comme lue(s)`);
   296→    }
   297→  };
   298→
   299→  const toggleDigest = async () => {
   300→    if (!prefs) return;
   301→    const res = await api<{ prefs: typeof prefs }>("/api/notifications", {
   302→      body: { action: "prefs", digestEnabled: !prefs.digestEnabled },
   303→    });
   304→    if (res.ok) {
   305→      setPrefs(res.prefs);
   306→      toast.success(res.prefs.digestEnabled ? "Résumé quotidien 19h activé" : "Résumé quotidien désactivé");
   307→    }
   308→  };
   309→
   310→  return (
   311→    <div className="space-y-4">
   312→      <FcCard className="p-5">
   313→        <div className="mb-3 flex items-center justify-between">
   314→          <div className="flex items-center gap-2">
   315→            <Bell className="h-5 w-5 text-[#C89F62]" />
   316→            <h3 className="fc-serif text-lg font-bold text-[#1E4D2B]">Centre d&apos;alertes</h3>
   317→          </div>
   318→          <button
   319→            onClick={markAll}
   320→            className="flex items-center gap-1 text-[12px] font-medium text-[#2A4A3A] hover:underline"
   321→          >
   322→            <CheckCheck className="h-4 w-4" /> Tout lire
   323→          </button>
   324→        </div>
   325→
   326→        {loading ? (
   327→          <p className="py-6 text-center text-sm text-[#8A9288]">Chargement…</p>
   328→        ) : items.length === 0 ? (
   329→          <p className="py-6 text-center text-sm text-[#8A9288]">
   330→            Aucune notification pour le moment. Les événements de la famille apparaîtront ici.
   331→          </p>
   332→        ) : (
   333→          <ul className="fc-scroll max-h-96 space-y-2 overflow-y-auto pr-1">
   334→            {items.map((n) => (
   335→              <li
   336→                key={n.id}
   337→                className={`rounded-2xl border p-3 transition-colors ${
   338→                  n.read ? "border-[#E5DCC0] bg-[#FFFDF6]" : "border-[#C89F62]/50 bg-[#C89F62]/10"
   339→                }`}
   340→              >
   341→                <div className="mb-1 flex items-start justify-between gap-2">
   342→                  <p className="text-[14px] font-bold text-[#1E4D2B]">{n.title}</p>
   343→                  <PriorityBadge priority={n.priority} />
   344→                </div>
   345→                <p className="text-[13px] leading-snug text-[#5C6B60]">{n.body}</p>
   346→                <div className="mt-2 flex items-center justify-between">
   347→                  <span className="text-[11px] text-[#8A9288]">
   348→                    {n.univers} · {timeAgo(n.createdAt)}
   349→                  </span>
   350→                  {!n.read && (
   351→                    <button
   352→                      onClick={() => markRead(n.id)}
   353→                      className="flex items-center gap-1 text-[11px] font-medium text-[#2A4A3A] hover:underline"
   354→                    >
   355→                      <Check className="h-3 w-3" /> Marquer lue
   356→                    </button>
   357→                  )}
   358→                </div>
   359→              </li>
   360→            ))}
   361→          </ul>
   362→        )}
   363→      </FcCard>
   364→
   365→      {prefs && (
   366→        <FcCard className="p-5">
   367→          <h3 className="fc-serif mb-3 text-lg font-bold text-[#1E4D2B]">Préférences</h3>
   368→          <label className="flex items-center justify-between gap-3 text-[14px] text-[#2B3530]">
   369→            <span>
   370→              Résumé quotidien à 19 h
   371→              <span className="block text-[12px] text-[#8A9288]">Anti-saturation : 1 récapitulatif au lieu de multiples alertes</span>
   372→            </span>
   373→            <input
   374→              type="checkbox"
   375→              checked={prefs.digestEnabled}
   376→              onChange={toggleDigest}
   377→              className="h-5 w-5 accent-[#1F5B3F]"
   378→            />
   379→          </label>
   380→          <p className="mt-3 rounded-xl bg-[#3D6B52]/5 p-3 text-[12px] leading-relaxed text-[#5C6B60]">
   381→            🔴 Les alertes <strong>critiques</strong> (décès, caisse en déficit) contournent toujours le silence
   382→            nocturne et le résumé — conformément au règlement de la famille.
   383→          </p>
   384→        </FcCard>
   385→      )}
   386→    </div>
   387→  );
   388→}

   394→function PushCard() {
   395→  const [state, setState] = useState<PushState>("default");
   396→  const [busy, setBusy] = useState(false);
   397→  const [isIOS] = useState(() =>
   398→    typeof navigator !== "undefined" &&
   399→    (/iPhone|iPad|iPod/.test(navigator.userAgent) ||
   400→      (navigator.platform === "MacIntel" && navigator.maxTouchPoints > 1))
   401→  );
   402→
   403→  useEffect(() => {
   404→    const t = setTimeout(async () => setState(await getPushState()), 0);
   405→    return () => clearTimeout(t);
   406→  }, []);
   407→
   408→  const enable = async () => {
   409→    setBusy(true);
   410→    const ok = await enablePush();
   411→    if (ok) setState(await getPushState());
   412→    setBusy(false);
   413→  };
   414→
   415→  const verify = async () => {
   416→    setBusy(true);
   417→    const res = await testPush();
   418→    if (res.ok) {
   419→      toast.success("Notification envoyée — regardez votre écran, même vérrouillé 🔔");
   420→    } else {
   421→      toast.error(res.error || "Envoi impossible.");
   422→    }
   423→    setBusy(false);
   424→  };
   425→
   426→  return (
   427→    <FcCard className="p-5">
   428→      <div className="mb-2 flex items-center gap-2">
   429→        <Bell className="h-5 w-5 text-[#C89F62]" />
   430→        <h3 className="fc-serif text-lg font-bold text-[#1E4D2B]">Notifications du téléphone</h3>
   431→      </div>
   432→      {state === "unsupported" && (
   433→        <p className="text-[13px] leading-relaxed text-[#8A9288]">
   434→          {isIOS
   435→            ? "Sur iPhone : ajoutez d'abord l'app à l'écran d'accueil (Partager → Sur l'écran d'accueil), puis ouvrez-la depuis l'icône."
   436→            : "Ce navigateur ne prend pas en charge les notifications push."}
   437→        </p>
   438→      )}
   439→      {state === "denied" && (
   440→        <p className="flex items-center gap-2 text-[13px] text-[#B91C1C]">
   441→          <BellOff className="h-4 w-4" /> Notifications bloquées — réautorisez-les dans les réglages du site
   442→          (icône cadenas dans la barre d'adresse).
   443→        </p>
   444→      )}
   445→      {(state === "default" || state === "granted") && (
   446→        <>
   447→          <p className="mb-3 text-[13px] leading-relaxed text-[#5C6B60]">
   448→            Recevez les événements de la famille même application fermée — avec le son et la vibration,
   449→            comme WhatsApp. Gratuit et illimité.
   450→          </p>
   451→          <FcButton onClick={enable} disabled={busy}>
   452→            {busy ? "Activation…" : "Activer les notifications"}
   453→          </FcButton>
   454→        </>
   455→      )}
   456→      {state === "subscribed" && (
   457→        <>
   458→          <p className="mb-3 flex items-center gap-1.5 text-[13px] font-medium text-[#16A34A]">
   459→            <ShieldCheck className="h-4 w-4" /> Actives sur cet appareil — même app fermée.
