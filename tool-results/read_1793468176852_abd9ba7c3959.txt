   103→function makeCode(): string {
   104→  const chars = "ABCDEFGHJKMNPQRSTUVWXYZ23456789";
   105→  const pick = (n: number) =>
   106→    Array.from({ length: n }, () => chars[Math.floor(Math.random() * chars.length)]).join("");
   107→  return `FC-${pick(4)}-${pick(4)}`;
   108→}
   109→// Clé VAPID factice (65 octets base64url) — permet au navigateur d'accepter la souscription
   110→function fakeVapid(): string {
   111→  const bytes = new Uint8Array(65);
   112→  bytes[0] = 4;
   113→  for (let i = 1; i < 65; i++) bytes[i] = (i * 7) % 256;
   114→  let bin = "";
   115→  bytes.forEach((b) => (bin += String.fromCharCode(b)));
   116→  return btoa(bin).replace(/\+/g, "-").replace(/\//g, "_").replace(/=+$/, "");
   117→}

   124→// Identifiant : 3 à 30 caractères — lettres, chiffres, point, tiret, underscore
   125→function normalizeIdentifier(raw: string): string | null {
   126→  const id = String(raw ?? "").trim().toLowerCase();
   127→  if (!/^[a-z0-9._-]{3,30}$/.test(id)) return null;
   128→  return id;

   131→// Hash mot de passe (mode démo) : SHA-256 salé via WebCrypto, format sha$salt$hash
   132→async function hashPassword(password: string, salt?: string): Promise<string> {
   133→  const s = salt || Math.random().toString(36).slice(2, 10) + Date.now().toString(36);
   134→  const msg = new TextEncoder().encode(`fc:${s}:${password}`);
   135→  try {
   136→    const digest = await crypto.subtle.digest("SHA-256", msg);
   137→    const hex = Array.from(new Uint8Array(digest), (b) => b.toString(16).padStart(2, "0")).join("");
   138→    return `sha$${s}$${hex}`;
   139→  } catch {
   140→    // Repli sans WebCrypto (rare) — FNV-1a 32 bits salé
   141→    let h = 2166136261;
   142→    const str = `fc:${s}:${password}`;
   143→    for (let i = 0; i < str.length; i++) {
   144→      h ^= str.charCodeAt(i);
   145→      h = Math.imul(h, 16777619);
   146→    }
   147→    return `fnv$${s}$${(h >>> 0).toString(16)}`;
   148→  }
   149→}
   150→function verifyPassword(password: string, stored: string | undefined): boolean {
   151→  if (!stored) return false;
   152→  const [, salt, hash] = stored.split("$");
   153→  if (!salt || !hash) return false;
   154→  // Vérification SYNCHRONE : recalcul du même schéma que hashPassword
   155→  const str = `fc:${salt}:${password}`;
   156→  let h = 2166136261;
   157→  for (let i = 0; i < str.length; i++) {
   158→    h ^= str.charCodeAt(i);
   159→    h = Math.imul(h, 16777619);
   160→  }
   161→  // Le hash WebCrypto (64 hex) ne peut PAS être recalculé en synchrone :
   162→  // on stocke donc TOUJOURS le hash via hashPassword (async) et on vérifie
   163→  // en re-hasant de manière asynchrone côté appelant (checkPassword).
   164→  return hash.length === 64 ? false : (h >>> 0).toString(16) === hash;
   165→}
   166→async function checkPassword(password: string, stored: string | undefined): Promise<boolean> {
   167→  if (!stored) return false;
   168→  const [, salt, hash] = stored.split("$");
   169→  if (!salt || !hash) return false;
   170→  if (hash.length === 64) {
   171→    const recomputed = await hashPassword(password, salt);
   172→    return recomputed === stored;
   173→  }
   174→  return verifyPassword(password, stored);
   175→}

   177→function seedDB(): DemoDB {
   178→  const now = nowISO();
   179→  // ── AUCUNE donnée fictive ──
   180→  // La base démarre VIDE : chaque membre, notification et chiffre affiché
   181→  // provient uniquement d'actions réelles (inscriptions, événements…).
   182→  return {
   183→    users: [],
   184→    members: [],
   185→    devices: [],
   186→    notifications: [],
   187→    logs: [
   188→      { id: rid("l"), userId: null, action: "SEED_EMPTY", detail: "Base initialisée vide — aucune donnée fictive", createdAt: now },
   189→    ],
   190→    prefs: { digestEnabled: true, quietStart: 19, quietEnd: 7, pushEnabled: true },
   191→    session: null,
   192→    pendingVerify: null,
   193→    pendingRegister: null,
   194→  };
   195→}
   196→
   197→function db(): DemoDB {
   198→  if (typeof localStorage === "undefined") return seedDB();
   199→  try {
   200→    const raw = localStorage.getItem(DB_KEY);
   201→    if (raw) return JSON.parse(raw) as DemoDB;
   202→  } catch {
   203→    // base corrompue → re-seed
   204→  }
   205→  const fresh = seedDB();
   206→  localStorage.setItem(DB_KEY, JSON.stringify(fresh));
   207→  return fresh;
   208→}
   209→function save(d: DemoDB): void {
   210→  if (typeof localStorage !== "undefined") localStorage.setItem(DB_KEY, JSON.stringify(d));
   211→}

   213→function currentUser(d: DemoDB): DemoUser | null {
   214→  if (!d.session) return null;
   215→  return d.users.find((u) => u.id === d.session!.uid) ?? null;
   216→}
   217→
   218→function notifyOne(d: DemoDB, userId: string, n: Partial<DemoNotification> & { title: string; body: string }) {
   219→  d.notifications.unshift({
   220→    id: rid("n"), userId, univers: n.univers ?? "U1", typeCode: n.typeCode ?? "INFO",
   221→    priority: n.priority ?? "NORMALE", title: n.title, body: n.body, read: false, createdAt: nowISO(),
   222→  });
   223→}
   224→function logOne(d: DemoDB, userId: string | null, action: string, detail?: string) {
   225→  d.logs.unshift({ id: rid("l"), userId, action, detail: detail ?? null, createdAt: nowISO() });
   226→}

   230→async function handleApi(url: string, method: string, body: Record<string, unknown>): Promise<Record<string, unknown>> {
   231→  const path = url.split("?")[0];
   232→  const d = db();
   233→
   234→  // ── Auth : session ──
   235→  // ── Géolocalisation d'indicatif : en démo, pas d'IP Cloudflare →
   236→  //    le client utilise ses replis (fuseau horaire, langue, Cameroun).
   237→  if (path === "/api/geo" && method === "GET") {
   238→    return { ok: true, country: null };
   239→  }
   240→
   241→  if (path === "/api/auth/session" && method === "GET") {
   242→    const user = currentUser(d);
   243→    if (!user) return { ok: true, session: null };
   244→    const device = d.devices.find((x) => x.id === d.session!.did && !x.revoked) ?? null;
   245→    const member = d.members.find((m) => m.id === user.memberId) ?? null;
   246→    const location =
   247→      member && member.locationConfirmed
   248→        ? {
   249→            country: member.country ?? null,
   250→            region: member.region ?? null,
   251→            department: member.department ?? null,
   252→            city: member.city ?? null,
   253→            arrondissement: member.arrondissement ?? null,
   254→            quartier: member.quartier ?? null,
   255→            latitude: member.latitude ?? null,
   256→            longitude: member.longitude ?? null,
   257→          }
   258→        : null;
   259→    return {
   260→      ok: true,
   261→      session: {
   262→        user: {
   263→          id: user.id, fullName: user.fullName, phone: user.phone, email: null,
   264→          role: user.role, hasPin: !!user.pinHash, hasLocation: !!location,
   265→        },
   266→        member: member
   267→          ? { id: member.id, branch: member.branch, role: member.role, status: member.status }
   268→          : null,
   269→        location,
   270→        device: device ? { id: device.id, name: device.name, type: device.type } : null,
   271→        unreadCount: d.notifications.filter((n) => n.userId === user.id && !n.read).length,
   272→        hasPasskey: false,
   273→      },
   274→    };
   275→  }
   276→
   277→  // ── Auth : flux numéro façon WhatsApp ──
   278→  if (path === "/api/auth/phone" && method === "POST") {
   279→    if (body.action === "request") {
   280→      const phone = normalizePhone(String(body.phone ?? ""), body.country);
   281→      if (!phone) return { ok: false, error: phoneErrorHint(body.country) };
   282→      const code = String(Math.floor(100000 + Math.random() * 900000));
   283→      const verifyToken = rid("vt");
   284→      d.pendingVerify = { phone, code, verifyToken };
   285→      const known = d.users.some((u) => u.phone === phone);
   286→      logOne(d, null, "PHONE_VERIFY_REQUEST", `${phone} — compte existant : ${known ? "oui" : "non"}`);
   287→      save(d);
   288→      return { ok: true, verifyToken, demoCode: code, known, phone };
   289→    }
   290→    if (body.action === "verify") {
   291→      const pv = d.pendingVerify;
   292→      if (!pv || pv.verifyToken !== body.verifyToken) {
   293→        return { ok: false, error: "Notification de vérification expirée. Recommencez avec votre numéro." };
   294→      }
   295→      if (String(body.code ?? "") !== pv.code) {
   296→        return { ok: false, error: "Code incorrect. Vérifiez la notification reçue." };
   297→      }
   298→      const user = d.users.find((u) => u.phone === pv.phone);
   299→      if (user) {
   300→        const device: DemoDevice = {
   301→          id: rid("dev"), userId: user.id, name: String(body.deviceName ?? "Appareil").slice(0, 60),
   302→          type: "pwa_web", simulated: false, hasPasskey: false, hasPush: false,
   303→          lastSeen: nowISO(), revoked: false, createdAt: nowISO(),
   304→        };
   305→        d.devices.push(device);
   306→        d.session = { uid: user.id, did: device.id };
   307→        notifyOne(d, user.id, {
   308→          typeCode: "AUTH_PHONE_LOGIN", title: "Connexion par numéro",
   309→          body: `Vous êtes connecté avec le numéro ${pv.phone} sur « ${device.name} ». Si ce n'était pas vous, signalez-le au secrétaire de famille.`,
   310→        });
   311→        logOne(d, user.id, "PHONE_LOGIN", `${pv.phone} — appareil « ${device.name} »`);
   312→        d.pendingVerify = null;
   313→        save(d);
   314→        return { ok: true, registered: true, user: { id: user.id, fullName: user.fullName } };
   315→      }
   316→      const registerToken = rid("reg");
   317→      d.pendingRegister = { phone: pv.phone, registerToken };
   318→      d.pendingVerify = null;
   319→      save(d);
   320→      return { ok: true, registered: false, registerToken, phone: pv.phone };
   321→    }
   322→    if (body.action === "register") {
   323→      const pr = d.pendingRegister;
   324→      if (!pr || pr.registerToken !== body.registerToken) {
   325→        return { ok: false, error: "Vérification expirée. Recommencez avec votre numéro." };
   326→      }
   327→      const firstName = String(body.firstName ?? "").trim();
   328→      const lastName = String(body.lastName ?? "").trim();
   329→      const pseudo = String(body.pseudo ?? "").trim().slice(0, 60);
   330→      if (!((firstName.length >= 2 && lastName.length >= 2) || pseudo.length >= 2)) {
   331→        return { ok: false, error: "Indiquez votre prénom et votre nom — ou au moins un pseudo." };
   332→      }
   333→      const fullName = pseudo || `${firstName} ${lastName}`;
   334→      const existing = d.users.find((u) => u.phone === pr.phone);
   335→      if (existing) {
   336→        const device: DemoDevice = {
   337→          id: rid("dev"), userId: existing.id, name: String(body.deviceName ?? "Appareil").slice(0, 60),
   338→          type: "pwa_web", simulated: false, hasPasskey: false, hasPush: false,
   339→          lastSeen: nowISO(), revoked: false, createdAt: nowISO(),
   340→        };
   341→        d.devices.push(device);
   342→        d.session = { uid: existing.id, did: device.id };
   343→        logOne(d, existing.id, "PHONE_LOGIN", `${pr.phone} — appareil « ${device.name} »`);
   344→        d.pendingRegister = null;
   345→        save(d);
   346→        return { ok: true, user: { id: existing.id, fullName: existing.fullName } };
   347→      }
   348→      const branch = BRANCHES[0];
   349→      const memberId = rid("m");
   350→      d.members.push({
   351→        id: memberId, fullName, phone: pr.phone, role: "ADULTE", status: "ACTIF",
   352→        branch: branch.name, branchId: branch.id, activeCode: null, codeExpiresAt: null, createdAt: nowISO(),
   353→      });
   354→      const user: DemoUser = { id: rid("u"), memberId, fullName, phone: pr.phone, role: "ADULTE" };
   355→      d.users.push(user);
   356→      const device: DemoDevice = {
   357→        id: rid("dev"), userId: user.id, name: String(body.deviceName ?? "Appareil").slice(0, 60),
   358→        type: "pwa_web", simulated: false, hasPasskey: false, hasPush: false,
   359→        lastSeen: nowISO(), revoked: false, createdAt: nowISO(),
   360→      };
   361→      d.devices.push(device);
   362→      d.session = { uid: user.id, did: device.id };
   363→      notifyOne(d, user.id, {
   364→        typeCode: "AUTH_WELCOME", title: "Bienvenue dans Famille.Mbeutetsa 🌳",
   365→        body: `Votre compte est actif (${branch.name}). Votre numéro de téléphone est votre identifiant : la prochaine fois, l'application s'ouvrira directement sur cet appareil.`,
   366→      });
   367→      // Notifier les « admins » de démo
   368→      d.users
   369→        .filter((u) => ["SECRETAIRE", "PATRIARCHE", "CHEF_BRANCHE", "CONSEIL"].includes(u.role))
   370→        .forEach((a) =>
   371→          notifyOne(d, a.id, {
   372→            typeCode: "AUTH_REGISTERED_ADMIN", title: "Nouveau membre inscrit",
   373→            body: `${fullName} (${branch.name}) vient de s'inscrire avec le numéro ${pr.phone}.`,
   374→          })
   375→        );
   376→      logOne(d, user.id, "REGISTERED", `${fullName} — ${pr.phone} — appareil « ${device.name} »`);
   377→      d.pendingRegister = null;
   378→      save(d);
   379→      return { ok: true, user: { id: user.id, fullName } };
   380→    }
   381→  }
   382→
   383→  // ── Auth : inscription / connexion par identifiant + mot de passe ──
   384→  if (path === "/api/auth/identifier" && method === "POST") {
   385→    const action = String(body.action ?? "");
   386→
   387→    if (action === "register") {
   388→      const identifier = normalizeIdentifier(String(body.identifier ?? ""));
   389→      const password = String(body.password ?? "");
   390→      const pseudo = String(body.pseudo ?? "").trim().slice(0, 60);
   391→      const phone = normalizePhone(String(body.phone ?? ""), body.country);
   392→      const deviceName = String(body.deviceName ?? "Appareil").slice(0, 60);
   393→
   394→      if (!identifier) return { ok: false, error: "Identifiant invalide : 3 à 30 caractères (lettres, chiffres, point, tiret ou underscore)." };
   395→      if (password.length < 6) return { ok: false, error: "Mot de passe trop court : au moins 6 caractères." };
   396→      if (pseudo.length < 2) return { ok: false, error: "Indiquez votre pseudo (au moins 2 caractères)." };
   397→      if (!phone) return { ok: false, error: phoneErrorHint(body.country) };
   398→
   399→      // 1. Identifiant déjà pris ?
   400→      if (d.users.some((u) => u.identifier === identifier)) {
   401→        return { ok: false, error: "Cet identifiant est déjà utilisé. Connectez-vous ou choisissez-en un autre." };
   402→      }
   403→
   404→      // 2. DÉTECTION par numéro de téléphone
   405→      const detected = d.users.find((u) => u.phone === phone);
   406→      if (detected) {
   407→        if (detected.identifier && detected.identifier !== identifier) {
   408→          return { ok: false, error: `Ce numéro est déjà associé au compte « ${detected.fullName} ». Connectez-vous avec son identifiant, ou utilisez un autre numéro.` };
   409→        }
   410→        // Compte détecté sans identifiant → association identifiant + mot de passe
   411→        detected.identifier = identifier;
   412→        detected.passwordHash = await hashPassword(password);
   413→        const device: DemoDevice = {
   414→          id: rid("dev"), userId: detected.id, name: deviceName, type: "pwa_web",
   415→          simulated: false, hasPasskey: false, hasPush: false,
   416→          lastSeen: nowISO(), revoked: false, createdAt: nowISO(),
   417→        };
   418→        d.devices.push(device);
   419→        d.session = { uid: detected.id, did: device.id };
   420→        notifyOne(d, detected.id, {
   421→          typeCode: "AUTH_PASSWORD_LOGIN", title: "Identifiant associé à votre compte",
   422→          body: `Votre identifiant « ${identifier} » est désormais relié au numéro ${phone}. Vous pouvez vous connecter avec ou sans mot de passe.`,
   423→        });
   424→        logOne(d, detected.id, "IDENTIFIER_LINKED", `${detected.fullName} — identifiant ${identifier} — ${phone}`);

   430→      // 3. Nouveau compte complet

   435→        branch: branch.name, branchId: branch.id, activeCode: null, codeExpiresAt: null, createdAt: nowISO(),
   436→      });
   437→      const user: DemoUser = {
   438→        id: rid("u"), memberId, fullName: pseudo, phone, role: "ADULTE",
   439→        identifier, passwordHash: await hashPassword(password),
   440→      };
   441→      d.users.push(user);
   442→      const device: DemoDevice = {
   443→        id: rid("dev"), userId: user.id, name: deviceName, type: "pwa_web",
   444→        simulated: false, hasPasskey: false, hasPush: false,

   453→      d.users

   456→          notifyOne(d, a.id, {

   459→          })
   460→        );
   461→      logOne(d, user.id, "REGISTERED_IDENTIFIER", `${pseudo} — identifiant ${identifier} — ${phone} — appareil « ${deviceName} »`);
   462→      d.pendingVerify = null;
   463→      d.pendingRegister = null;
   464→      save(d);
   465→      return { ok: true, linked: false, user: { id: user.id, fullName: pseudo } };
   466→    }
   467→
   468→    if (action === "login") {
   469→      const rawId = String(body.identifier ?? "").trim();
   470→      const password = String(body.password ?? "");
   471→      const deviceName = String(body.deviceName ?? "Appareil").slice(0, 60);
   472→      if (!rawId || !password) {
   473→        return { ok: false, error: "Indiquez votre identifiant et votre mot de passe." };
   474→      }
   475→      const phone = normalizePhone(rawId);
   476→      const identifier = phone ? null : normalizeIdentifier(rawId);
   477→      const user = phone
   478→        ? d.users.find((u) => u.phone === phone && u.passwordHash)
   479→        : identifier
   480→          ? d.users.find((u) => u.identifier === identifier)
   481→          : undefined;
   482→      if (!user || !(await checkPassword(password, user.passwordHash))) {
   483→        logOne(d, user?.id ?? null, "LOGIN_IDENTIFIER_FAILED", "identifiant ou mot de passe incorrect");
   484→        save(d);
   485→        return { ok: false, error: "Identifiant ou mot de passe incorrect." };
   486→      }
   487→      const device: DemoDevice = {
   488→        id: rid("dev"), userId: user.id, name: deviceName, type: "pwa_web",
   489→        simulated: false, hasPasskey: false, hasPush: false,
   490→        lastSeen: nowISO(), revoked: false, createdAt: nowISO(),
   491→      };
   492→      d.devices.push(device);
   493→      d.session = { uid: user.id, did: device.id };
   494→      notifyOne(d, user.id, {
   495→        typeCode: "AUTH_PASSWORD_LOGIN", title: "Connexion par identifiant",
   496→        body: `Vous êtes connecté avec l'identifiant « ${user.identifier ?? user.phone} » sur « ${device.name} ». Si ce n'était pas vous, signalez-le au secrétaire de famille.`,
   497→      });
   498→      logOne(d, user.id, "LOGIN_IDENTIFIER", `${user.fullName} — appareil « ${device.name} »`);
   499→      save(d);
   500→      return { ok: true, user: { id: user.id, fullName: user.fullName } };
   501→    }
   502→  }
   503→
   504→  // ── Auth : logout ──
   505→  if (path === "/api/auth/logout" && method === "POST") {
   506→    if (d.session) logOne(d, d.session.uid, "LOGOUT", "Session fermée (mode démo)");
   507→    d.session = null;
   508→    save(d);
   509→    return { ok: true };
   510→  }

   514→    const user = currentUser(d);
   515→    return {
   516→      ok: true,
   517→      stats: {
   518→        members: d.members.length,

   529→    const user = currentUser(d);

   535→      location: {
   536→        country: member.country ?? null, region: member.region ?? null,
   537→        department: member.department ?? null, city: member.city ?? null,
   538→        arrondissement: member.arrondissement ?? null, quartier: member.quartier ?? null,
   539→        latitude: member.latitude ?? null, longitude: member.longitude ?? null,
   540→        accuracy: member.accuracy ?? null, source: member.locationSource ?? null,
   541→        updatedAt: member.locationUpdatedAt ?? null,
   542→      },
   543→    };
   544→  }
   545→  if (path === "/api/me/location" && method === "POST") {
   546→    const user = currentUser(d);
   547→    if (!user) return { ok: false, error: "Session expirée." };
   548→    const member = d.members.find((m) => m.id === user.memberId);
   549→    if (!member) return { ok: false, error: "Aucun profil membre relié à ce compte." };
   550→    const s = (v: unknown, max = 80): string | null => {
   551→      const t = String(v ?? "").trim().slice(0, max);
   552→      return t.length ? t : null;
   553→    };
   554→    const lat = typeof body.latitude === "number" && isFinite(body.latitude) ? Math.max(-90, Math.min(90, body.latitude)) : null;
   555→    const lon = typeof body.longitude === "number" && isFinite(body.longitude) ? Math.max(-180, Math.min(180, body.longitude)) : null;
   556→    const country = s(body.country);
   557→    const city = s(body.city);
   558→    if (!country && !city && (lat === null || lon === null)) {
   559→      return { ok: false, error: "Indiquez au moins votre pays ou votre ville (ou activez le GPS)." };
   560→    }
   561→    member.country = country;
   562→    member.region = s(body.region);
   563→    member.department = s(body.department);
   564→    member.city = city;
   565→    member.arrondissement = s(body.arrondissement);
   566→    member.quartier = s(body.quartier);
   567→    member.latitude = lat;
   568→    member.longitude = lon;
   569→    member.accuracy = typeof body.accuracy === "number" && isFinite(body.accuracy) ? body.accuracy : null;
   570→    member.locationSource = s(body.source, 12);
   571→    member.locationConfirmed = true;
   572→    member.locationUpdatedAt = nowISO();
   573→    logOne(d, user.id, "LOCATION_SAVED", [member.city, member.region, member.country].filter(Boolean).join(", "));
   574→    save(d);
   575→    return { ok: true, location: { /* reflété par /api/auth/session */ } };
   576→  }
   577→
   578→  // ── Annuaire localisé de la famille ──
   579→  if (path === "/api/family/locations" && method === "GET") {
   580→    const user = currentUser(d);
   581→    if (!user) return { ok: false, error: "Session expirée." };
   582→    const out = d.members
   583→      .filter((m) => m.locationConfirmed && typeof m.latitude === "number" && typeof m.longitude === "number")
   584→      .map((m) => ({
   585→        id: m.id,
   586→        fullName: m.fullName,
   587→        phone: m.phone,
   588→        isMe: m.id === user.memberId,
   589→        role: m.role,
   590→        city: m.city ?? null,
   591→        region: m.region ?? null,
   592→        department: m.department ?? null,
   593→        arrondissement: m.arrondissement ?? null,
   594→        quartier: m.quartier ?? null,
   595→        country: m.country ?? null,
   596→        latitude: m.latitude as number,
   597→        longitude: m.longitude as number,
   598→        updatedAt: m.locationUpdatedAt ?? null,
   599→      }));
   600→    return { ok: true, members: out };

   602→
   603→  // ── Push ── (le Worker Cloudflare fc-push envoie les vraies notifications système)
   604→  const PUSH_URL = process.env.NEXT_PUBLIC_FC_PUSH_URL || "";
   605→  if (path === "/api/push/public-key" && method === "GET") {
   606→    if (PUSH_URL) {
   607→      try {
   608→        const r = await fetch(`${PUSH_URL}/public-key`);
   609→        const j = (await r.json()) as { publicKey?: string };
   610→        if (j.publicKey) return { ok: true, publicKey: j.publicKey };
   611→      } catch {
   612→        /* secours ci-dessous */
   613→      }
   614→    }
   615→    return { ok: true, publicKey: fakeVapid() };
   616→  }
   617→  if (path === "/api/push/subscribe" && method === "POST") {
   618→    const sub = body.subscription as { endpoint?: string } | undefined;
   619→    if (d.session && sub?.endpoint) {
   620→      const dev = d.devices.find((x) => x.id === d.session!.did);
   621→      if (dev) {
   622→        dev.hasPush = true;
   623→        dev.pushSubscription = JSON.stringify(sub);
   624→      }
   625→      save(d);
   626→      if (PUSH_URL) {
   627→        void fetch(`${PUSH_URL}/subscribe`, {
   628→          method: "POST",
   629→          headers: { "Content-Type": "application/json" },
   630→          body: JSON.stringify({ subscription: sub, phone: currentUser(d)?.phone }),
   631→        }).catch(() => undefined);
   632→      }
   633→    }
   634→    return { ok: true, subscribed: true };
   635→  }
   636→  if (path === "/api/push" && method === "POST") {
   637→    const user = currentUser(d);
   638→    const action = String(body.action ?? "");
   639→    if ((action === "test" || action === "welcome") && user) {
   640→      const dev = d.session ? d.devices.find((x) => x.id === d.session!.did) : null;
   641→      const subJson = dev?.pushSubscription;
   642→      const welcome = action === "welcome";
   643→      if (PUSH_URL && subJson) {
   644→        try {
   645→          const r = await fetch(`${PUSH_URL}/send-one`, {
   646→            method: "POST",
   647→            headers: { "Content-Type": "application/json" },
   648→            body: JSON.stringify({
   649→              subscription: JSON.parse(subJson),
   650→              title: welcome ? "🌳 Bienvenue dans la famille !" : "🔔 Famille.Mbeutetsa — test réussi",
   651→              body: welcome
   652→                ? `Félicitations ${user.fullName} — les alertes de la famille vous parviendront ici, même l'application fermée.`
   653→                : `Les notifications push fonctionnent sur « ${dev!.name} ». Vous ne manquerez plus aucun événement familial.`,
   654→              tag: welcome ? "fc-welcome" : "fc-push-test",
   655→            }),
   656→          });
   657→          if (r.ok) {
   658→            logOne(d, user.id, welcome ? "PUSH_WELCOME_SENT" : "PUSH_TEST_SENT", "Notification système réelle (Worker)");
   659→            save(d);
   660→            return { ok: true, sent: true };
   661→          }
   662→        } catch {
   663→          /* secours in-app ci-dessous */
   664→        }
   665→      }
   666→      notifyOne(d, user.id, {
   667→        typeCode: welcome ? "AUTH_WELCOME" : "PUSH_TEST", priority: "NORMALE",
   668→        title: welcome ? "Bienvenue dans Famille.Mbeutetsa 🌳" : "Notification de test",
   669→        body: welcome
   670→          ? "Votre compte est actif. Activez les notifications dans Ma sécurité pour être alerté sur l'écran du téléphone."
   671→          : "Le serveur de notification n'est pas encore relié — cette alerte reste dans le centre d'alertes.",
   672→      });
   673→      logOne(d, user.id, welcome ? "PUSH_WELCOME_INAPP" : "PUSH_TEST_INAPP", "Notification in-app (secours)");
   674→      save(d);
   675→      return { ok: true, sent: false, fallback: "in-app" };
   676→    }

   691→      if (body.action === "read") {

   693→        if (n) n.read = true;
   694→        save(d);
   695→        return { ok: true };
   696→      }
   697→      if (body.action === "read-all") {
   698→        let count = 0;
   699→        d.notifications.forEach((n) => {
   700→          if (n.userId === user.id && !n.read) {
   701→            n.read = true;
   702→            count++;
   703→          }
   704→        });
   705→        save(d);
   706→        return { ok: true, readAll: count };
   707→      }
   708→      if (body.action === "prefs") {
   709→        d.prefs = { ...d.prefs, digestEnabled: !!body.digestEnabled };
   710→        save(d);
   711→        return { ok: true, prefs: d.prefs };
   712→      }
   713→    }
   714→  }
   715→
   716→  // ── Appareils ──
   717→  if (path === "/api/devices" && method === "GET") {
   718→    const user = currentUser(d);
   719→    if (!user) return { ok: false, error: "Session expirée." };
   720→    return {
   721→      ok: true,
   722→      devices: d.devices
   723→        .filter((x) => x.userId === user.id)
   724→        .map((x) => ({
   725→          id: x.id, name: x.name, type: x.type, simulated: x.simulated, hasPasskey: x.hasPasskey,
   726→          hasPush: x.hasPush, lastSeen: x.lastSeen, revoked: x.revoked,
   727→          current: !!d.session && x.id === d.session.did, createdAt: x.createdAt,
   728→        })),
   729→    };
   730→  }
   731→  if (path === "/api/devices" && method === "POST" && body.action === "revoke") {
   732→    const dev = d.devices.find((x) => x.id === body.id);
   733→    if (dev) {
   734→      dev.revoked = true;
   735→      logOne(d, dev.userId, "DEVICE_REVOKED", `Appareil « ${dev.name} » révoqué`);
   736→      save(d);
   737→    }
   738→    return { ok: true };
   739→  }
   740→
   741→  // ── Journal de sécurité ──
   742→  if (path === "/api/security-log" && method === "GET") {
   743→    const user = currentUser(d);
   744→    if (!user) return { ok: false, error: "Session expirée." };
   745→    return {
   746→      ok: true,
   747→      logs: d.logs.filter((l) => l.userId === user.id || l.userId === null).slice(0, 50),
   748→    };
   749→  }
   750→
   751→  // ── Secrétariat (admin) ──
   752→  if (path === "/api/admin/members") {
   753→    if (method === "GET") {
   754→      return { ok: true, members: d.members, branches: BRANCHES };
   755→    }
   756→    if (method === "POST") {
   757→      const fullName = String(body.fullName ?? "").trim();
   758→      if (fullName.length < 3) return { ok: false, error: "Nom complet requis." };
   759→      const branch = BRANCHES.find((b) => b.id === body.branchId) ?? BRANCHES[0];
   760→      const code = makeCode();
   761→      d.members.push({
   762→        id: rid("m"), fullName, phone: String(body.phone ?? ""), email: String(body.email ?? ""),
   763→        role: String(body.role ?? "ADULTE"), status: "PREINSCRIT", branch: branch.name, branchId: branch.id,
   764→        activeCode: code, codeExpiresAt: new Date(Date.now() + 7 * 86400e3).toISOString(), createdAt: nowISO(),
   765→      });
   766→      logOne(d, null, "MEMBER_PREREGISTERED", `${fullName} (${branch.name}) — code ${code}`);
   767→      save(d);
   768→      return { ok: true, member: { fullName }, accessCode: { code } };
   769→    }
   770→    if (method === "PUT") {
   771→      const m = d.members.find((x) => x.id === body.memberId);
   772→      if (!m) return { ok: false, error: "Membre introuvable." };
   773→      const code = makeCode();
   774→      m.activeCode = code;
   775→      m.codeExpiresAt = new Date(Date.now() + 7 * 86400e3).toISOString();
   776→      logOne(d, null, "CODE_REGENERATED", `Nouveau code pour ${m.fullName}`);
   777→      save(d);
   778→      return { ok: true, accessCode: { code } };
   779→    }
   780→  }
   781→
   782→  return { ok: false, error: `Route démo non simulée : ${method} ${path}` };
   783→}

   793→export function installDemoApi(): void {
   794→  if (typeof window === "undefined" || window.__FC_DEMO_API__) return;
   795→  window.__FC_DEMO_API__ = true;
   796→  const originalFetch = window.fetch.bind(window);
   797→  window.fetch = async (input: RequestInfo | URL, init?: RequestInit): Promise<Response> => {
   798→    const url = typeof input === "string" ? input : input instanceof URL ? input.toString() : input.url;
   799→    if (!url.startsWith("/api/")) return originalFetch(input as RequestInfo, init);
   800→
   801→    // ── Notifications push & événements : fetch RÉEL (même origine) ──
   802→    // La passerelle push (_worker.js sur pages.dev / routes Node en local) gère
   803→    // /api/push/* et /api/notify*. Seul /api/push/subscribe est intercepté pour
   804→    // marquer l'appareil comme « push actif » dans la base de démo, puis relayé.
   805→    if (url === "/api/push/subscribe" && (init?.method || "POST").toUpperCase() === "POST") {
   806→      let subBody: Record<string, unknown> = {};
   807→      try {
   808→        subBody = JSON.parse(String(init?.body || "{}")) as Record<string, unknown>;
   809→      } catch {
   810→        subBody = {};
   811→      }
   812→      const sub = subBody.subscription as { endpoint?: string } | undefined;
   813→      const d = db();
   814→      if (d.session && sub?.endpoint) {
   815→        const dev = d.devices.find((x) => x.id === d.session!.did);
   816→        if (dev) {
   817→          dev.hasPush = true;
   818→          dev.pushSubscription = JSON.stringify(sub);
   819→        }
   820→        save(d);
   821→      }
   822→      // Relais réel (originalFetch passe au-dessus de ce hook) → passerelle push
   823→      return originalFetch("/api/push/subscribe", {
   824→        method: "POST",
   825→        headers: { "Content-Type": "application/json" },
   826→        body: String(init?.body || "{}"),
   827→      });
   828→    }
   829→    if (url === "/api/push" || url.startsWith("/api/push/") || url.startsWith("/api/notify")) {
   830→      return originalFetch(input as RequestInfo, init);
   831→    }
   832→
   833→    const method = (init?.method || (init?.body ? "POST" : "GET")).toUpperCase();
   834→    let body: Record<string, unknown> = {};
   835→    try {
   836→      if (init?.body) body = JSON.parse(String(init.body));
   837→    } catch {
   838→      body = {};
   839→    }
   840→    const data = await handleApi(url, method, body);
   841→    return new Response(JSON.stringify(data), {
   842→      status: data.ok === false ? 400 : 200,
   843→      headers: { "Content-Type": "application/json" },
   844→    });
   845→  };
   846→}
